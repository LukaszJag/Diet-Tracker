Here is the complete implementation of the hierarchical categories and
subcategories system for your notes feature.

Step 1: Azure SQL Database Migration

To store both the main category and the subcategory, run this script once in
your Azure Portal Query Editor (or Azure Data Studio / SSMS) on database
free-sql-db-6758529:

-- Add subcategory column to the notes table if it does not already exist
IF NOT EXISTS (
    SELECT * FROM sys.columns 
    WHERE object_id = OBJECT_ID(N'[diet_tracker_schema].[notes]') 
    AND name = 'subcategory'
)
BEGIN
    ALTER TABLE [diet_tracker_schema].[notes]
    ADD [subcategory] NVARCHAR(255) NULL;
END

Explanation:
Existing notes table stores [category], which will now hold the Main Category
(e.g., "Java ecosystem"). The new [subcategory] column stores the Subcategory
(e.g., "Spring"). Adding it as NVARCHAR(255) NULL ensures backward compatibility
with notes that were already created without subcategories.

package com.lukaszjag.diet_tracker_android.tools.sql_tools;

public class QueryMaker {
    public static String getAllProductTableLikeProductNameCaseInsensitive(String productData) {
        String escapedData = productData != null ? productData.replace("'", "''") : "";
        return "SELECT * FROM [diet_tracker_schema].[product_table] WHERE LOWER(product_name) LIKE '%" + escapedData + "%'";
    }

    public static String getAllProductsFromCalendarTableLikeDate(String date) {
        String escapedDate = date != null ? date.replace("'", "''") : "";
        return "SELECT * FROM [diet_tracker_schema].[calendar] WHERE day_date LIKE '%" + escapedDate + "%'";
    }

    // --- NOTES QUERIES ---

    public static String getAllNotesQuery() {
        return "SELECT * FROM [diet_tracker_schema].[notes]";
    }

    public static String insertNoteQuery(String title, String subtitle, String description,
                                         String category, String subcategory,
                                         boolean isUrgently, boolean isLearning, boolean isGeneral,
                                         boolean isToday, boolean isToBuy, String created, String deadline) {

        String safeTitle = title != null ? title.replace("'", "''") : "";
        String safeSub = subtitle != null ? subtitle.replace("'", "''") : "";
        String safeDesc = description != null ? description.replace("'", "''") : "";
        String safeCat = category != null ? category.replace("'", "''") : "";
        String safeSubcat = subcategory != null ? subcategory.replace("'", "''") : "";

        return "INSERT INTO [diet_tracker_schema].[notes] " +
                "([title], [subtitle], [description], [category], [subcategory], [is_urgently], [is_learning], " +
                "[is_general_to_do], [is_today_task], [is_to_buy_task], [date_created], [date_deadline]) " +
                "VALUES (" +
                "N'" + safeTitle + "', " +
                "N'" + safeSub + "', " +
                "N'" + safeDesc + "', " +
                "N'" + safeCat + "', " +
                "N'" + safeSubcat + "', " +
                (isUrgently ? 1 : 0) + ", " +
                (isLearning ? 1 : 0) + ", " +
                (isGeneral ? 1 : 0) + ", " +
                (isToday ? 1 : 0) + ", " +
                (isToBuy ? 1 : 0) + ", " +
                "'" + created + "', " +
                (deadline == null || deadline.isEmpty() ? "NULL" : "'" + deadline + "'") +
                ");";
    }

    public static String updateNoteQuery(long id, String title, String subtitle, String description,
                                         String category, String subcategory,
                                         boolean isUrgently, boolean isLearning, boolean isGeneral,
                                         boolean isToday, boolean isToBuy, String deadline) {

        String safeTitle = title != null ? title.replace("'", "''") : "";
        String safeSub = subtitle != null ? subtitle.replace("'", "''") : "";
        String safeDesc = description != null ? description.replace("'", "''") : "";
        String safeCat = category != null ? category.replace("'", "''") : "";
        String safeSubcat = subcategory != null ? subcategory.replace("'", "''") : "";

        return "UPDATE [diet_tracker_schema].[notes] SET " +
                "[title] = N'" + safeTitle + "', " +
                "[subtitle] = N'" + safeSub + "', " +
                "[description] = N'" + safeDesc + "', " +
                "[category] = N'" + safeCat + "', " +
                "[subcategory] = N'" + safeSubcat + "', " +
                "[is_urgently] = " + (isUrgently ? 1 : 0) + ", " +
                "[is_learning] = " + (isLearning ? 1 : 0) + ", " +
                "[is_general_to_do] = " + (isGeneral ? 1 : 0) + ", " +
                "[is_today_task] = " + (isToday ? 1 : 0) + ", " +
                "[is_to_buy_task] = " + (isToBuy ? 1 : 0) + ", " +
                "[date_deadline] = " + (deadline == null || deadline.isEmpty() ? "NULL" : "'" + deadline + "'") + " " +
                "WHERE [id] = " + id + ";";
    }
}

Explanation:
insertNoteQuery and updateNoteQuery now accept the subcategory argument and
append [subcategory] = N'...' into the SQL statements. Existing single quotes
are escaped with replace("'", "''") to prevent SQL syntax errors.
getAllNotesQuery() runs SELECT *, which automatically fetches the new
subcategory column.

package com.lukaszjag.diet_tracker_android.tools.notes_tool;

import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.CheckBox;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;

import com.lukaszjag.diet_tracker_android.R;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class MyAdapter extends RecyclerView.Adapter<MyAdapter.MyViewHolder> {

    public static final int SORT_NONE = 0;
    public static final int SORT_DATE_NEWEST = 1;
    public static final int SORT_DATE_OLDEST = 2;
    public static final int SORT_URGENCY_HIGH = 3;
    public static final int SORT_URGENCY_LOW = 4;

    public interface OnItemClickListener {
        void onItemClick(int position, Note note);
    }

    private List<Note> noteList;
    private List<Note> originalList;
    private OnItemClickListener listener;

    private boolean hideSection1 = false;
    private boolean hideSection2 = false;
    private boolean hideSection3 = false;
    private boolean hideSection4 = false;

    private int sortCriteria = SORT_NONE;

    public MyAdapter() {
        this.noteList = new ArrayList<>();
        this.originalList = new ArrayList<>();
    }

    public List<Note> getOriginalList() { return originalList; }

    public void setOnItemClickListener(OnItemClickListener listener) { this.listener = listener; }

    public void setSectionVisibilities(boolean hideS1, boolean hideS2, boolean hideS3, boolean hideS4) {
        this.hideSection1 = hideS1;
        this.hideSection2 = hideS2;
        this.hideSection3 = hideS3;
        this.hideSection4 = hideS4;
        notifyDataSetChanged();
    }

    public void setSortCriteria(int criteria) {
        this.sortCriteria = criteria;
        applySort();
        notifyDataSetChanged();
    }

    public void addItem(Note note) {
        noteList.add(note);
        originalList.add(note);
        applySort();
        notifyItemInserted(noteList.indexOf(note));
    }

    public void setItem(int position, Note note) {
        if (position >= 0 && position < noteList.size()) {
            Note oldNote = noteList.get(position);
            int origIdx = originalList.indexOf(oldNote);
            if (origIdx != -1) originalList.set(origIdx, note);
            noteList.set(position, note);
            applySort();
            notifyDataSetChanged();
        }
    }

    public void deleteItem(int position) {
        if (position >= 0 && position < noteList.size()) {
            Note note = noteList.get(position);
            originalList.remove(note);
            noteList.remove(position);
            notifyItemRemoved(position);
            notifyItemRangeChanged(position, noteList.size());
        }
    }

    private void applySort() {
        if (sortCriteria == SORT_NONE) return;
        Collections.sort(noteList, (n1, n2) -> {
            switch (sortCriteria) {
                case SORT_DATE_NEWEST:
                    return (n2.getDateCreated() != null ? n2.getDateCreated() : "")
                            .compareTo(n1.getDateCreated() != null ? n1.getDateCreated() : "");
                case SORT_DATE_OLDEST:
                    return (n1.getDateCreated() != null ? n1.getDateCreated() : "")
                            .compareTo(n2.getDateCreated() != null ? n2.getDateCreated() : "");
                case SORT_URGENCY_HIGH:
                    return Integer.compare(getUrgencyWeight(n2), getUrgencyWeight(n1));
                case SORT_URGENCY_LOW:
                    return Integer.compare(getUrgencyWeight(n1), getUrgencyWeight(n2));
                default:
                    return 0;
            }
        });
    }

    private int getUrgencyWeight(Note note) {
        if (note == null) return 0;
        int idx = note.getUrgentScaleEnglish().indexOf(note.getNoteUrgently());
        return (idx != -1) ? idx : 0;
    }

    /**
     * Filters notes by subtitle, main category, subcategory, and status flags.
     * - If main category is "All Categories", all categories match.
     * - If a main category is selected and subcategory is "All Subcategories", all notes
     *   with that main category match.
     * - If both are selected, both must match.
     */
    public void filter(String subtitle, String selectedMainCategory, String selectedSubcategory,
                       boolean showLearningOnly, boolean showGeneralOnly, boolean showTodayOnly, boolean showToBuyOnly) {
        noteList.clear();
        String qSub = (subtitle != null) ? subtitle.toLowerCase().trim() : "";
        boolean isAllCategories = (selectedMainCategory == null || selectedMainCategory.equals("All Categories"));
        boolean isAllSubcategories = (selectedSubcategory == null || selectedSubcategory.equals("All Subcategories"));

        for (Note note : originalList) {
            boolean matchSub = qSub.isEmpty() || (note.getNoteSubtitle() != null && note.getNoteSubtitle().toLowerCase().contains(qSub));

            // Category and Subcategory matching
            boolean matchCat = true;
            if (!isAllCategories) {
                matchCat = note.getNoteCategory() != null && note.getNoteCategory().equalsIgnoreCase(selectedMainCategory.trim());
                if (matchCat && !isAllSubcategories) {
                    matchCat = note.getNoteSubcategory() != null && note.getNoteSubcategory().equalsIgnoreCase(selectedSubcategory.trim());
                }
            }

            boolean matchLearning = !showLearningOnly || note.isLearning();
            boolean matchGeneral = !showGeneralOnly || note.isGeneralToDo();
            boolean matchToday = !showTodayOnly || note.isTodayTask();
            boolean matchToBuy = !showToBuyOnly || note.isToBuyTask();

            if (matchSub && matchCat && matchLearning && matchGeneral && matchToday && matchToBuy) {
                noteList.add(note);
            }
        }
        applySort();
        notifyDataSetChanged();
    }

package com.lukaszjag.diet_tracker_android.gui.notes;

            @Override
            public void onSuccess(ArrayList<RowInTable> resultTable) {
                runOnUiThread(() -> {
                    adapter.clearAll();
                    for (RowInTable row : resultTable) {

                                row.getValue("title"),
                                row.getValue("subtitle"),
                                row.getValue("description"),

                                row.getValue("is_urgently"),
                                "true".equals(row.getValue("is_learning")) || "1".equals(row.getValue("is_learning")),

    private void showNoteDialog(Note note) {
        AlertDialog.Builder builder = new AlertDialog.Builder(this);
        View dialogView = getLayoutInflater().inflate(R.layout.dialog_note, null);
        builder.setView(dialogView);

        final EditText title = dialogView.findViewById(R.id.dialogTitle);
        final EditText subtitle = dialogView.findViewById(R.id.dialogSubtitle);
        final EditText desc = dialogView.findViewById(R.id.dialogDescription);
        final Spinner spinnerCategory = dialogView.findViewById(R.id.dialogSpinnerCategory);
        final Spinner spinnerSubcategory = dialogView.findViewById(R.id.dialogSpinnerSubcategory);
        final CheckBox cbLearn = dialogView.findViewById(R.id.dialogCbLearning);
        final CheckBox cbGen = dialogView.findViewById(R.id.dialogCbGeneral);
        final CheckBox cbToday = dialogView.findViewById(R.id.dialogCbToday);
        final CheckBox cbBuy = dialogView.findViewById(R.id.dialogCbToBuy);

        boolean isEditMode = (note != null);

        // Populate Main Categories Spinner
        List<String> mainCategories = NoteCategoryData.getMainCategories();
        ArrayAdapter<String> catAdapter = new ArrayAdapter<>(this, android.R.layout.simple_spinner_item, mainCategories);
        catAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        spinnerCategory.setAdapter(catAdapter);

        // When Main Category changes in Dialog, update Subcategory Spinner
        spinnerCategory.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                String chosenMain = mainCategories.get(position);
                List<String> subcategories = NoteCategoryData.getSubcategories(chosenMain);
                List<String> displaySubs = new ArrayList<>();
                if (subcategories.isEmpty()) {
                    displaySubs.add("None");
                } else {
                    displaySubs.addAll(subcategories);
                }

                ArrayAdapter<String> subAdapter = new ArrayAdapter<>(Notes.this, android.R.layout.simple_spinner_item, displaySubs);
                subAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
                spinnerSubcategory.setAdapter(subAdapter);

                // If in Edit mode and subcategory belongs to this main category, pre-select it
                if (isEditMode && note.getNoteSubcategory() != null) {
                    int subIndex = displaySubs.indexOf(note.getNoteSubcategory());
                    if (subIndex != -1) {
                        spinnerSubcategory.setSelection(subIndex);
                    }
                }
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {}
        });

        // Pre-fill fields if editing
        if (isEditMode) {
            title.setText(note.getNoteTitle());
            subtitle.setText(note.getNoteSubtitle());
            desc.setText(note.getNoteDescription());

            if (note.getNoteCategory() != null) {
                int catIndex = mainCategories.indexOf(note.getNoteCategory());

                    spinnerCategory.setSelection(catIndex);

                        title.getText().toString(),
                        subtitle.getText().toString(),
                        desc.getText().toString(),
                        selectedMain,
                        selectedSub,
                        false,
                        cbLearn.isChecked(),
                        cbGen.isChecked(),
                        cbToday.isChecked(),
                        cbBuy.isChecked(),
                        null
                );

                        title.getText().toString(),
                        subtitle.getText().toString(),
                        desc.getText().toString(),
                        selectedMain,
                        selectedSub,
                        false,
                        cbLearn.isChecked(),
                        cbGen.isChecked(),
                        cbToday.isChecked(),
                        cbBuy.isChecked(),
                        currentDate,
                        null
                );
            }
