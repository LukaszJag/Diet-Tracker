import androidx.appcompat.app.AppCompatActivity;

import com.google.android.material.datepicker.MaterialDatePicker;
import com.lukaszjag.diet_tracker_android.R;
import com.lukaszjag.diet_tracker_android.tools.cloud_data_tools.AzureDataCallback;
import com.lukaszjag.diet_tracker_android.tools.cloud_data_tools.GetFromSQLDatabase;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Date;
import java.util.List;
import java.util.Locale;
import java.util.TimeZone;

public class GetProductData extends AppCompatActivity {

    //<editor-fold desc="Global variables">
    ArrayList<RowInTable> products = new ArrayList<>();
    int counter = 0;
    String productComment = "none";

    //<editor-fold desc="UI components">
    private Spinner productSpinner;
    private Spinner mealNameSpinner;

    //<editor-fold desc="Button">
    private Button getProductFromDatabase;
    private Button previousButton;
    private Button nextButton;
    private Button dateDataButton;

    private Button productCommentButton;
    private Button acceptButton;
    //</editor-fold>

    //<editor-fold desc="EditText">
    private EditText productNameEditText;
    private EditText weightEditText;
    //</editor-fold>

    //<editor-fold desc="TextView">
    private TextView productNameTextView;
    private TextView brandLabel;
    private TextView kcalLabel;
    private TextView proteinLabel;
    private TextView fatLabel;
    private TextView carbsLabel;
    private TextView brandValueTextView;
    private TextView kcalValueTextView;
    private TextView proteinValueTextView;
    private TextView fatValueTextView;
    private TextView carbsValueTextView;

    private TextView dateDataTextView;
    //</editor-fold>

    //<editor-fold desc="Gather Information From UI - Variables">
    String dayDate;
    String dayName;
    String mealName;
    double amountOfProduct;
    String productName;
    String productBrand = "";
    String timeOptional;
    String commentOptional;

    //<editor-fold desc="Macro variables">
    double kcal;
    double protein;
    double fat;
    double carbs;

    double kcalConsume;
    double carbsConsume;
    double proteinConsume;
    double fatConsume;

    String rawKcal;
    String rawProtein;
    String rawFat;
    String rawCarbs;
    //</editor-fold>
    //</editor-fold>

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        this.getWindow().setFlags(
                WindowManager.LayoutParams.FLAG_FULLSCREEN,
                WindowManager.LayoutParams.FLAG_FULLSCREEN
        );

        setContentView(R.layout.get_product_data);
        setupAllElements();
    }

    private void setupAllElements() {
        initUIComponents();
        setupComponents();
        addListeners();
    }

    private void setupComponents() {
        ArrayList<String> mealNamesList = new ArrayList<>(Arrays.asList(
                "None", "Breakfast", "Second Breakfast", "Snack 1",
                "Dinner", "Snack 2", "Supper", "After workout", "Night snack"
        ));

        ArrayAdapter<String> adapter = new ArrayAdapter<>(
                GetProductData.this,
                android.R.layout.simple_spinner_item,
                mealNamesList
        );
        adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        mealNameSpinner.setAdapter(adapter);
    }

    private void initUIComponents() {
        productSpinner = findViewById(R.id.productSpinner);
        mealNameSpinner = findViewById(R.id.mealNameSpinner);

        getProductFromDatabase = findViewById(R.id.getProductFromDatabase);
        previousButton = findViewById(R.id.previousButton);
        nextButton = findViewById(R.id.nextButton);
        dateDataButton = findViewById(R.id.datePickButton);

        productCommentButton = findViewById(R.id.productCommentButton);
        acceptButton = findViewById(R.id.acceptButton);

        productNameEditText = findViewById(R.id.productNameEditText);
        weightEditText = findViewById(R.id.weightEditText);

        productNameTextView = findViewById(R.id.productNameTextView);
        brandLabel = findViewById(R.id.brandLabel);
        kcalLabel = findViewById(R.id.kcalLabel);
        proteinLabel = findViewById(R.id.proteinLabel);
        fatLabel = findViewById(R.id.fatLabel);

        brandValueTextView = findViewById(R.id.brandValueTextView);
        kcalValueTextView = findViewById(R.id.kcalValueTextView);
        proteinValueTextView = findViewById(R.id.proteinValueTextView);
        fatValueTextView = findViewById(R.id.fatValueTextView);
        carbsValueTextView = findViewById(R.id.carbsValueTextView);

        dateDataTextView = findViewById(R.id.dateDataTextView);
        dateDataTextView.setText(MyDate.getCurrentDayInSQLFormat());
    }

    private void setValuesToMacroUITable() {
        brandValueTextView.setText(productBrand != null ? productBrand : "-");
        kcalValueTextView.setText(String.format(Locale.getDefault(), "%.0f", kcal));
        proteinValueTextView.setText(String.format(Locale.getDefault(), "%.1f", protein));
        fatValueTextView.setText(String.format(Locale.getDefault(), "%.1f", fat));
        carbsValueTextView.setText(String.format(Locale.getDefault(), "%.1f", carbs));
    }

    private void getMacroValues() {
        if (products.isEmpty() || counter < 0 || counter >= products.size()) {
            return;
        }

        dayDate = dateDataTextView.getText().toString();
        dayName = MyDate.getDayNameInLowerCase(dayDate);

        if (mealNameSpinner.getSelectedItem() != null) {
            mealName = mealNameSpinner.getSelectedItem().toString();
        }

        // Fetch brand from product row if available, otherwise fallback
        String fetchedBrand = products.get(counter).getValue("brand_name");
        if (fetchedBrand == null) {
            fetchedBrand = products.get(counter).getValue("product_brand");
        }
        productBrand = (fetchedBrand != null) ? fetchedBrand : "";

        productComment = products.get(counter).getValue("comment_optional");
        if (productComment == null) productComment = "none";

        if (productSpinner.getSelectedItem() != null) {
            productName = productSpinner.getSelectedItem().toString();
        }

        // 1. Safe parsing of weight
        String amountOfProductTmp = weightEditText.getText().toString().trim();
        try {
            if (!amountOfProductTmp.isEmpty()) {
                amountOfProduct = Double.parseDouble(amountOfProductTmp);
            } else {
                amountOfProduct = 0.0;
            }
        } catch (NumberFormatException e) {
            amountOfProduct = 0.0;
        }

        // 2. Safe parsing of macros
        try {
            rawKcal = products.get(counter).getValue("product_kcal");
            kcal = Double.parseDouble(rawKcal);
        } catch (NumberFormatException | NullPointerException e) {
            kcal = 0.0;
        }

        try {
            rawProtein = products.get(counter).getValue("product_protein");
            protein = Double.parseDouble(rawProtein);
        } catch (NumberFormatException | NullPointerException e) {
            protein = 0.0;
        }

        try {
            rawFat = products.get(counter).getValue("product_fat");
            fat = Double.parseDouble(rawFat);
        } catch (NumberFormatException | NullPointerException e) {
            fat = 0.0;
        }

        try {
            rawCarbs = products.get(counter).getValue("product_carbs");
            carbs = Double.parseDouble(rawCarbs);
        } catch (NumberFormatException | NullPointerException e) {
            carbs = 0.0;
        }
    }

    private void showSimpleDialog() {
        // Prevent IndexOutOfBoundsException when no products are loaded
        if (products.isEmpty() || counter < 0 || counter >= products.size()) {
            Toast.makeText(GetProductData.this, "No product selected to view comments.", Toast.LENGTH_SHORT).show();
            return;
        }

        new AlertDialog.Builder(GetProductData.this)
                .setTitle(products.get(counter).getValue("product_name"))
                .setMessage(productComment != null ? productComment : "No comment")
                .setPositiveButton("OK", (dialog, which) -> dialog.dismiss())
                .show();
    }

    private void gatherInformationFromUIPrintValues() {
        Log.d("DIET_DEBUG", "dayDate: " + dayDate);
        Log.d("DIET_DEBUG", "dayName: " + dayName);
        Log.d("DIET_DEBUG", "mealName: " + mealName);
        Log.d("DIET_DEBUG", "productBrand: " + productBrand);
        Log.d("DIET_DEBUG", "productComment: " + productComment);
        Log.d("DIET_DEBUG", "productName: " + productName);
        Log.d("DIET_DEBUG", "amountOfProduct: " + amountOfProduct);
        Log.d("DIET_DEBUG", "kcal: " + kcal + " | protein: " + protein + " | fat: " + fat + " | carbs: " + carbs);
    }

    private void addListeners() {
        productCommentButton.setOnClickListener(v -> showSimpleDialog());

        getProductFromDatabase.setOnClickListener(v -> {
            String searchText = productNameEditText.getText().toString().trim();
            String query = QueryMaker.getAllProductTableLikeProductNameCaseInsensitive(searchText);

            GetFromSQLDatabase.runAzureQueryAIChat(query, new AzureDataCallback() {
                @Override
                public void onSuccess(ArrayList<RowInTable> resultTable) {
                    // Always perform UI updates on the Main UI thread
                    runOnUiThread(() -> {
                        products = resultTable;
                        counter = 0;

                        Toast.makeText(GetProductData.this, "Found " + products.size() + " products", Toast.LENGTH_SHORT).show();

                        List<String> productList = new ArrayList<>();
                        for (int i = 0; i < products.size(); i++) {
                            productList.add(products.get(i).getValue("product_name"));
                        }

                        ArrayAdapter<String> adapter = new ArrayAdapter<>(
                                GetProductData.this,
                                android.R.layout.simple_spinner_item,
                                productList
                        );
                        adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
                        productSpinner.setAdapter(adapter);

                        if (!products.isEmpty()) {
                            productNameTextView.setText(products.get(0).getValue("product_name"));
                            getMacroValues();
                            setValuesToMacroUITable();
                        }
                    });
                }

                @Override
                public void onFailure(String errorMessage) {
                    runOnUiThread(() -> {
                        Toast.makeText(GetProductData.this, "Failed to get data: " + errorMessage, Toast.LENGTH_SHORT).show();
                    });
                }
            });
        });

        nextButton.setOnClickListener(view -> {
            if (!products.isEmpty()) {
                if (counter < products.size() - 1) {
                    counter++;
                } else {
                    counter = 0;
                }
                productSpinner.setSelection(counter);
                getMacroValues();
                setValuesToMacroUITable();
            }
        });

        previousButton.setOnClickListener(view -> {
            if (!products.isEmpty()) {
                if (counter > 0) {
                    counter--;
                } else {
                    counter = products.size() - 1;
                }
                productSpinner.setSelection(counter);
                getMacroValues();
                setValuesToMacroUITable();
            }
        });

        dateDataButton.setOnClickListener(v -> {
            MaterialDatePicker<Long> datePicker = MaterialDatePicker.Builder.datePicker()
                    .setTitleText("Select a Date")
                    .setSelection(MaterialDatePicker.todayInUtcMilliseconds())
                    .build();

            datePicker.addOnPositiveButtonClickListener(selection -> {
                // Ensure date formatting uses UTC to prevent 1-day timezone shifting
                SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault());
                sdf.setTimeZone(TimeZone.getTimeZone("UTC"));
                String formattedDate = sdf.format(new Date(selection));

                dateDataTextView.setText(formattedDate);
            });

            datePicker.show(getSupportFragmentManager(), "MATERIAL_DATE_PICKER");
        });

        productSpinner.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                if (!products.isEmpty() && position < products.size()) {
                    counter = position;
                    productNameTextView.setText(products.get(counter).getValue("product_name"));
                    getMacroValues();
                    setValuesToMacroUITable();
                }
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {}
        });

        acceptButton.setOnClickListener(v -> {
            getMacroValues();

            if (productName == null || productName.trim().isEmpty()) {
                Toast.makeText(GetProductData.this, "Please select a product first", Toast.LENGTH_SHORT).show();
                return;
            }

            if (amountOfProduct <= 0) {
                Toast.makeText(GetProductData.this, "Please enter a valid weight", Toast.LENGTH_SHORT).show();
                return;
            }

            // Consumed macros based on standard 100g product nutrition values
            kcalConsume = (kcal * amountOfProduct) / 100.0;
            carbsConsume = (carbs * amountOfProduct) / 100.0;
            fatConsume = (fat * amountOfProduct) / 100.0;
            proteinConsume = (protein * amountOfProduct) / 100.0;

            timeOptional = "none";
            commentOptional = (productComment != null && !productComment.trim().isEmpty()) ? productComment : "none";

            gatherInformationFromUIPrintValues();

            String myInsertQuery = QueryMaker.insertMealToCalendarTable(
                    dayDate,
                    dayName,
                    mealName,
                    amountOfProduct,
                    productName,
                    kcal,
                    protein,
                    fat,
                    carbs,
                    timeOptional,
                    commentOptional,
                    kcalConsume,
                    carbsConsume,
                    fatConsume,
                    proteinConsume
            );

            Log.d("AZURE_INSERT_SQL", "Executing: " + myInsertQuery);

            GetFromSQLDatabase.runAzureQueryAIChat(myInsertQuery, new AzureDataCallback() {
                @Override
                public void onSuccess(ArrayList<RowInTable> resultTable) {
                    runOnUiThread(() -> {
                        Toast.makeText(GetProductData.this, "Meal added successfully!", Toast.LENGTH_SHORT).show();
                        weightEditText.setText("");
                        DietWidgetProvider.forceWidgetUpdate(GetProductData.this);
                    });
                }

                @Override
                public void onFailure(String errorMessage) {
                    runOnUiThread(() -> {
                        Toast.makeText(GetProductData.this, "Failed to add meal: " + errorMessage, Toast.LENGTH_LONG).show();
                        Log.e("AZURE_INSERT_ERROR", errorMessage);
                    });
                }
            });
        });
    }
}

            public void onClick(View v) {
                shiftDate(-1);
            }
        });

        nextDayButton.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                shiftDate(1);
            }
        });
    }

    private void shiftDate(int amount) {
        try {
            Date currentDate = sqlDateFormat.parse(dateOfData);
            if (currentDate != null) {
                Calendar cal = Calendar.getInstance();
                cal.setTime(currentDate);
                cal.add(Calendar.DATE, amount);
                dateOfData = sqlDateFormat.format(cal.getTime());
                updateDateAndFetch();
            }
        } catch (Exception e) {
            Log.e("DayData", "Error updating calendar selection offset: " + e.getMessage());
        }
    }

    private void updateDateAndFetch() {
        dateDataTextView.setText(dateOfData);
        updateDayOfWeekName(dateOfData);
        getDataForDate(dateOfData);
    }

    private void updateDayOfWeekName(String sqlDateStr) {
        try {
            Date date = sqlDateFormat.parse(sqlDateStr);
            if (date != null) {
                String nameOfDay = dayNameFormat.format(date);
                dayNameTextView.setText(nameOfDay);
            }
        } catch (Exception e) {
            dayNameTextView.setText("");
        }
    }

    private void getDataForDate(String dateInSQLFormat) {
        String query = QueryMaker.getAllProductsFromCalendarTableLikeDate(dateInSQLFormat);

        GetFromSQLDatabase.runAzureQueryAIChat(query, new AzureDataCallback() {
            @Override
            public void onSuccess(ArrayList<RowInTable> resultTable) {
                products = resultTable;
                setDataToView();
            }

            @Override
            public void onFailure(String errorMessage) {
                System.out.println("Failed to get data: " + errorMessage);
                Toast.makeText(DayData.this, "Connection Error: " + errorMessage, Toast.LENGTH_SHORT).show();
            }
        });
    }

    private void setDataToView() {
        adapter.clearItems();

        if (products == null || products.isEmpty()) {
            noDataTextView.setVisibility(View.VISIBLE);
            summaryLayout.setVisibility(View.GONE);
            recyclerView.setVisibility(View.GONE);
            return;
        }

        noDataTextView.setVisibility(View.GONE);
        summaryLayout.setVisibility(View.VISIBLE);
        recyclerView.setVisibility(View.VISIBLE);

        float totalKcal = 0;
        float totalProtein = 0;
        float totalFat = 0;
        float totalCarbs = 0;

        for (RowInTable row : products) {
            String productName = row.getValue("product_name");
            String mealName = row.getValue("meal_name");
            String amountOfProduct = row.getValue("amount_of_product");
            String kcalConsume = row.getValue("kcal_consume");
            String proteinConsume = row.getValue("protein_consume");
            String fatConsume = row.getValue("fat_consume");
            String carbsConsume = row.getValue("carbs_consume");
            String comment = row.getValue("comment_optional");

            float amount = safeParseFloat(amountOfProduct);
            float kcal = safeParseFloat(kcalConsume);
            float protein = safeParseFloat(proteinConsume);
            float fat = safeParseFloat(fatConsume);
            float carbs = safeParseFloat(carbsConsume);

            totalKcal += kcal;
            totalProtein += protein;
            totalFat += fat;
            totalCarbs += carbs;

            Product entry = new Product(productName, mealName, amount, kcal, protein, fat, carbs, comment);
            adapter.addItem(entry);
        }

        // Format all summary values to exactly two decimal places
        totalKcalTextView.setText(String.format(Locale.getDefault(), "Kcal: %.2f", totalKcal));
        totalProteinTextView.setText(String.format(Locale.getDefault(), "Prot: %.2fg", totalProtein));
        totalFatTextView.setText(String.format(Locale.getDefault(), "Fat: %.2fg", totalFat));
        totalCarbsTextView.setText(String.format(Locale.getDefault(), "Carb: %.2fg", totalCarbs));
    }

    private float safeParseFloat(String value) {
        if (value == null || value.trim().isEmpty() || value.equalsIgnoreCase("null")) {
            return 0f;
        }
        try {
            return Float.parseFloat(value);
        } catch (NumberFormatException e) {
            return 0f;
        }
    }

    private void showFullProductDataDialog(Product product) {
        AlertDialog.Builder builder = new AlertDialog.Builder(this);
        builder.setTitle(product.getProductName());

        StringBuilder text = new StringBuilder();
        text.append("Meal: ").append(product.getMealName() != null ? product.getMealName() : "N/A").append("\n");

        text.append("- Protein: ").append(product.getConsumedProtein()).append(" g\n");
        text.append("- Fat: ").append(product.getConsumedFat()).append(" g\n");
        text.append("- Carbohydrates: ").append(product.getConsumedCarbs()).append(" g\n\n");

        if (product.getCommentOptional() != null && !product.getCommentOptional().isEmpty() && !product.getCommentOptional().equalsIgnoreCase("null")) {
            text.append("Comment: ").append(product.getCommentOptional());
        }

        builder.show();
    }

        setContentView(R.layout.dialog_notes_calendar);

        if (getWindow() != null) {

    private void updateCalendar() {
        SimpleDateFormat sdf = new SimpleDateFormat("MMMM yyyy", Locale.getDefault());
        monthYearTextView.setText(sdf.format(currentDisplayCalendar.getTime()));

        // Calculate leading blanks (Monday-first: Mon=0 ... Sun=6)
        Calendar temp = (Calendar) currentDisplayCalendar.clone();
        temp.set(Calendar.DAY_OF_MONTH, 1);
        int firstDayOfWeek = temp.get(Calendar.DAY_OF_WEEK); // Sun=1, Mon=2
        int offset = (firstDayOfWeek == Calendar.SUNDAY) ? 6 : (firstDayOfWeek - Calendar.MONDAY);

        int daysInMonth = currentDisplayCalendar.getActualMaximum(Calendar.DAY_OF_MONTH);

        // Find max notes count in displayed month for proper gradient scaling
        int maxNotes = 1;
        int year = currentDisplayCalendar.get(Calendar.YEAR);
        int month = currentDisplayCalendar.get(Calendar.MONTH) + 1;
        for (int day = 1; day <= daysInMonth; day++) {
            String dateKey = String.format(Locale.getDefault(), "%04d-%02d-%02d", year, month, day);
            Integer count = notesCountPerDay.get(dateKey);
            if (count != null && count > maxNotes) {
                maxNotes = count;
            }
        }
        this.maxNotesCountInMonth = maxNotes;

        List<Integer> dayNumbers = new ArrayList<>();
        for (int i = 0; i < offset; i++) {
            dayNumbers.add(0); // empty cell placeholder
        }
        for (int day = 1; day <= daysInMonth; day++) {
            dayNumbers.add(day);
        }

        dayAdapter = new CalendarDayAdapter(dayNumbers);
        daysRecyclerView.setAdapter(dayAdapter);
        updateSelectedCountLabel();
    }

    private void updateSelectedCountLabel() {
        if (selectedDates.isEmpty()) {
            selectedCountTextView.setText("No days selected (Showing all notes)");
        } else {
            selectedCountTextView.setText("Selected: " + selectedDates.size() + " day(s)");
        }
    }

    // --- Grid Adapter for Days ---
    private class CalendarDayAdapter extends RecyclerView.Adapter<CalendarDayAdapter.DayViewHolder> {
        private final List<Integer> days;

        public CalendarDayAdapter(List<Integer> days) {
            this.days = days;
        }

        @NonNull
        @Override
        public DayViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
            View view = LayoutInflater.from(parent.getContext()).inflate(R.layout.item_calendar_day, parent, false);
            // Apply scale to cell height
            ViewGroup.LayoutParams lp = view.getLayoutParams();
            if (lp != null) {
                float density = getContext().getResources().getDisplayMetrics().density;
                lp.height = (int) (44 * density * uiScale);
                view.setLayoutParams(lp);
            }
            return new DayViewHolder(view);
        }

        @Override
        public void onBindViewHolder(@NonNull DayViewHolder holder, int position) {
            int dayNumber = days.get(position);
            if (dayNumber == 0) {
                holder.root.setVisibility(View.INVISIBLE);
                holder.root.setOnClickListener(null);
                return;
            }

            holder.root.setVisibility(View.VISIBLE);
            holder.tvDayNumber.setText(String.valueOf(dayNumber));
            holder.tvDayNumber.setTextSize(TypedValue.COMPLEX_UNIT_SP, 13 * uiScale);

            Calendar dayCal = (Calendar) currentDisplayCalendar.clone();
            dayCal.set(Calendar.DAY_OF_MONTH, dayNumber);
            boolean isSunday = (dayCal.get(Calendar.DAY_OF_WEEK) == Calendar.SUNDAY);

            // Sunday number MUST be dark red (#8B0000)
            if (isSunday) {
                holder.tvDayNumber.setTextColor(Color.parseColor("#8B0000"));
                holder.tvDayNumber.setTypeface(null, Typeface.BOLD);
            } else {
                holder.tvDayNumber.setTextColor(Color.parseColor("#222222"));
                holder.tvDayNumber.setTypeface(null, Typeface.NORMAL);
            }

            // Date string yyyy-MM-dd
            int year = currentDisplayCalendar.get(Calendar.YEAR);
            int month = currentDisplayCalendar.get(Calendar.MONTH) + 1;
            String dateKey = String.format(Locale.getDefault(), "%04d-%02d-%02d", year, month, dayNumber);

            // Note count heatmap calculation
            int noteCount = 0;
            if (notesCountPerDay.containsKey(dateKey)) {
                noteCount = notesCountPerDay.get(dateKey);
            }

            int backgroundColor;
            if (noteCount == 0) {
                // White if none
                backgroundColor = Color.WHITE;
            } else {
                // More intensive color as notes increase
                float intensity = 0.25f + 0.75f * ((float) noteCount / Math.max(1, maxNotesCountInMonth));
                backgroundColor = ColorUtils.blendARGB(Color.WHITE, heatmapBaseColor, Math.min(1.0f, intensity));
            }

            // Selection state
            boolean isSelected = selectedDates.contains(dateKey);
            GradientDrawable borderDrawable = new GradientDrawable();
            borderDrawable.setColor(backgroundColor);

            float density = getContext().getResources().getDisplayMetrics().density;
            if (isSelected) {
                borderDrawable.setStroke((int) (2.5f * density), Color.parseColor("#1565C0"));
                holder.ivCheck.setVisibility(View.VISIBLE);
            } else {
                borderDrawable.setStroke((int) (0.5f * density), Color.parseColor("#E0E0E0"));
                holder.ivCheck.setVisibility(View.GONE);
            }
            holder.root.setBackground(borderDrawable);

            // Toggle date selection on click
            holder.root.setOnClickListener(v -> {
                if (selectedDates.contains(dateKey)) {
                    selectedDates.remove(dateKey);
                } else {
                    selectedDates.add(dateKey);
                }
                notifyItemChanged(position);
                updateSelectedCountLabel();
            });
        }

        @Override
        public int getItemCount() {
            return days.size();
        }

        class DayViewHolder extends RecyclerView.ViewHolder {
            FrameLayout root;
            TextView tvDayNumber;
            TextView ivCheck;

            public DayViewHolder(@NonNull View itemView) {
                super(itemView);
                root = itemView.findViewById(R.id.calDayRoot);
                tvDayNumber = itemView.findViewById(R.id.calDayNumberTextView);
                ivCheck = itemView.findViewById(R.id.calDayCheckIndicator);
            }
        }

    public static void runAzureQueryAIChat(String mySqlString, AzureDataCallback callback) {
        AzureApiService apiService = RetrofitClient.getRetrofitInstance().create(AzureApiService.class);
        QueryRequest requestBody = new QueryRequest(mySqlString);

        Log.d("AZURE_DEBUG", "Sending Query: " + mySqlString);

        Call<List<Map<String, Object>>> call = apiService.executeCustomQuery(requestBody);

        call.enqueue(new Callback<List<Map<String, Object>>>() {
            @Override
            public void onResponse(Call<List<Map<String, Object>>> call, Response<List<Map<String, Object>>> response) {
                ArrayList<RowInTable> resultTable = new ArrayList<>();

                Log.d("AZURE_DEBUG", "Response Code: " + response.code());

                if (response.isSuccessful()) {
                    List<Map<String, Object>> dynamicSqlData = response.body();

                    if (dynamicSqlData != null && !dynamicSqlData.isEmpty()) {
                        for (int i = 0; i < dynamicSqlData.size(); i++) {
                            Map<String, Object> row = dynamicSqlData.get(i);
                            RowInTable rowInTable = new RowInTable();

                            for (String key : row.keySet()) {
                                rowInTable.putKeyAndValueToRow(key, String.valueOf(row.get(key)));
                            }
                            resultTable.add(rowInTable);
                        }
                        callback.onSuccess(resultTable);
                    } else {
                        Log.d("AZURE_DEBUG", "Query executed successfully with empty result set.");
                        callback.onSuccess(new ArrayList<>());
                    }
                } else {
                    String detailedError = "Unknown error";
                    try {
                        if (response.errorBody() != null) {
                            detailedError = response.errorBody().string();
                        }
                    } catch (IOException e) {
                        Log.e("AZURE_DEBUG", "Failed to parse error body: " + e.getMessage());
                    }

                    Log.e("AZURE_ERROR_DETAILS", "HTTP Error " + response.code() + ": " + detailedError);
                    callback.onFailure("Server error (" + response.code() + "): " + detailedError);
                }
            }

            @Override
            public void onFailure(Call<List<Map<String, Object>>> call, Throwable t) {
                Log.e("AZURE_DEBUG", "Network Transport Failure: " + t.getMessage(), t);
                callback.onFailure("Network fail: " + t.getMessage());
            }
        });
    }

package com.lukaszjag.diet_tracker_android.tools.cloud_data_tools;

import java.io.IOException;
import java.lang.annotation.Annotation;
import java.lang.reflect.Type;
import java.util.ArrayList;

import okhttp3.ResponseBody;
import retrofit2.Converter;
import retrofit2.Retrofit;

public class NullOnEmptyConverterFactory extends Converter.Factory {
    @Override
    public Converter<ResponseBody, ?> responseBodyConverter(Type type, Annotation[] annotations, Retrofit retrofit) {
        final Converter<ResponseBody, ?> delegate = retrofit.nextResponseBodyConverter(this, type, annotations);
        return new Converter<ResponseBody, Object>() {
            @Override
            public Object convert(ResponseBody body) throws IOException {
                if (body.contentLength() == 0) {
                    return new ArrayList<>();
                }
                String content = body.string();
                if (content == null || content.trim().isEmpty() || content.trim().equalsIgnoreCase("null")) {
                    return new ArrayList<>();
                }
                // Reconstruct the response body with the read content for Gson
                ResponseBody newBody = ResponseBody.create(body.contentType(), content);
                return delegate.convert(newBody);
            }
        };
    }
}

    private void applySort() {
        if (sortCriteria == SORT_NONE) return;
        Collections.sort(noteList, (n1, n2) -> {
            switch (sortCriteria) {
                case SORT_DATE_NEWEST:
                    return (n2.getDateCreated() != null ? n2.getDateCreated() : "")
                            .compareTo(n1.getDateCreated() != null ? n1.getDateCreated() : "");
                case SORT_DATE_OLDEST:
                    return (n1.getDateCreated() != null ? n1.getDateCreated() : "")
                            .compareTo(n2.getDateCreated() != null ? n2.getDateCreated() : "");
                case SORT_URGENCY_HIGH:
                    return Integer.compare(getUrgencyWeight(n2), getUrgencyWeight(n1));
                case SORT_URGENCY_LOW:
                    return Integer.compare(getUrgencyWeight(n1), getUrgencyWeight(n2));
                default:
                    return 0;
            }
        });
    }

    private int getUrgencyWeight(Note note) {
        if (note == null) return 0;
        int idx = note.getUrgentScaleEnglish().indexOf(note.getNoteUrgently());
        return (idx != -1) ? idx : 0;
    }

    public void filter(String subtitle, String selectedMainCategory, String selectedSubcategory,
                       boolean showLearningOnly, boolean showGeneralOnly, boolean showTodayOnly, boolean showToBuyOnly,
                       String statusFilter, Set<String> selectedDates) {
        noteList.clear();
        String qSub = (subtitle != null) ? subtitle.toLowerCase().trim() : "";
        boolean isAllCategories = (selectedMainCategory == null || selectedMainCategory.equals("All Categories"));
        boolean isAllSubcategories = (selectedSubcategory == null || selectedSubcategory.equals("All Subcategories"));

        boolean hasDateFilter = (selectedDates != null && !selectedDates.isEmpty());

        for (Note note : originalList) {
            boolean matchSub = qSub.isEmpty() || (note.getNoteSubtitle() != null && note.getNoteSubtitle().toLowerCase().contains(qSub));

            boolean matchCat = true;
            if (!isAllCategories) {
                matchCat = note.getNoteCategory() != null && note.getNoteCategory().equalsIgnoreCase(selectedMainCategory.trim());
                if (matchCat && !isAllSubcategories) {
                    matchCat = note.getNoteSubcategory() != null && note.getNoteSubcategory().equalsIgnoreCase(selectedSubcategory.trim());
                }
            }

            boolean matchLearning = !showLearningOnly || note.isLearning();
            boolean matchGeneral = !showGeneralOnly || note.isGeneralToDo();
            boolean matchToday = !showTodayOnly || note.isTodayTask();
            boolean matchToBuy = !showToBuyOnly || note.isToBuyTask();

            boolean matchStatus = true;
            if ("DONE".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDone();
            } else if ("DELETED".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDeleted();
            }

            boolean matchDate = true;
            if (hasDateFilter) {
                String created = note.getDateCreated();
                if (created != null && created.length() >= 10) {
                    created = created.substring(0, 10);
                }
                matchDate = (created != null && selectedDates.contains(created));
            }

            if (matchSub && matchCat && matchLearning && matchGeneral && matchToday && matchToBuy && matchStatus && matchDate) {
                noteList.add(note);
            }
        }
        applySort();
        notifyDataSetChanged();
    }

import android.content.Context;
import android.net.Uri;
import android.os.Environment;

import org.json.JSONArray;
import org.json.JSONObject;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;

public class NoteStorage {
    private static final String FILE_NAME = "notes_data.json";

    // Resolves dynamically to the standard Download/diet-tracker-data directory on any Android device
    private static final File TARGET_DIRECTORY = new File(
            Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS),
            "diet-tracker-data"
    );

    /**
     * Resolves the target directory and ensures the folder structure is created.
     */
    private static File getTargetFile() {
        if (!TARGET_DIRECTORY.exists()) {
            TARGET_DIRECTORY.mkdirs(); // Automatically creates "diet-tracker-data" if missing
        }
        return new File(TARGET_DIRECTORY, FILE_NAME);
    }

    /**
     * Saves the list of notes to the specified external storage path.
     */
    public static void saveNotes(Context context, List<Note> notes) {
        try {
            JSONArray jsonArray = new JSONArray();
            for (Note note : notes) {
                JSONObject noteJson = note.toJsonObject();
                if (noteJson != null) {
                    jsonArray.put(noteJson);
                }
            }
            String jsonContent = jsonArray.toString(4);

            File file = getTargetFile();
            try (FileOutputStream fos = new FileOutputStream(file);
                 OutputStreamWriter osw = new OutputStreamWriter(fos, StandardCharsets.UTF_8)) {
                osw.write(jsonContent);
            }
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error saving notes", e);
        }
    }

    /**
     * Loads the list of notes from the specified external storage path.
     */
    public static List<Note> loadNotes(Context context) {
        List<Note> notes = new ArrayList<>();
        File file = getTargetFile();

        if (!file.exists()) {
            return notes;
        }

        try (FileInputStream fis = new FileInputStream(file);
             InputStreamReader isr = new InputStreamReader(fis, StandardCharsets.UTF_8);
             BufferedReader reader = new BufferedReader(isr)) {

            StringBuilder sb = new StringBuilder();
            char[] buffer = new char[1024];
            int numRead;
            while ((numRead = reader.read(buffer)) != -1) {
                sb.append(buffer, 0, numRead);
            }

            String jsonContent = sb.toString();

            // Strip hidden UTF-8 BOM if present (often added by Windows/external text editors)
            if (jsonContent.startsWith("\uFEFF")) {
                jsonContent = jsonContent.substring(1);
            }
            jsonContent = jsonContent.trim();

            if (!jsonContent.isEmpty()) {
                JSONArray jsonArray = new JSONArray(jsonContent);
                for (int i = 0; i < jsonArray.length(); i++) {
                    JSONObject noteJson = jsonArray.getJSONObject(i);
                    Note note = Note.fromJsonObject(noteJson);
                    if (note != null) {
                        notes.add(note);
                    }
                }
            }
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error reading or parsing notes JSON file", e);
        }

        return notes;
    }

    /**
     * Imports external JSON files via Uri, validates structure, and overwrites local notes storage.
     */
    public static boolean importNotes(Context context, Uri sourceUri) {
        try (InputStream in = context.getContentResolver().openInputStream(sourceUri)) {
            if (in == null) return false;

            BufferedReader reader = new BufferedReader(new InputStreamReader(in, StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            char[] buffer = new char[1024];
            int numRead;
            while ((numRead = reader.read(buffer)) != -1) {
                sb.append(buffer, 0, numRead);
            }

            String jsonContent = sb.toString();

            // Strip hidden UTF-8 BOM if present
            if (jsonContent.startsWith("\uFEFF")) {
                jsonContent = jsonContent.substring(1);
            }
            jsonContent = jsonContent.trim();

            JSONArray jsonArray = new JSONArray(jsonContent);
            List<Note> importedNotes = new ArrayList<>();
            for (int i = 0; i < jsonArray.length(); i++) {
                Note note = Note.fromJsonObject(jsonArray.getJSONObject(i));
                if (note != null) {
                    importedNotes.add(note);
                }
            }

            // Write validated imported notes directly to the target external storage
            saveNotes(context, importedNotes);
            return true;
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error importing notes", e);
            return false;
        }
    }
}

        productList.clear();

public class ThemeHelper {

    /**
     * Traverses the current Activity's view hierarchy and updates all Button backgrounds.
     */
    public static void applyButtonTheme(Activity activity) {
        if (activity == null) return;
        int buttonColor = AppSettings.getButtonColor(activity);
        View root = activity.findViewById(android.R.id.content);
        if (root != null) {
            applyThemeToViewHierarchy(root, buttonColor);
        }
    }

    public static void applyThemeToViewHierarchy(View view, int color) {
        if (view == null) return;
        if (view instanceof Button) {
            applyColorToButton((Button) view, color);
        } else if (view instanceof ViewGroup) {
            ViewGroup vg = (ViewGroup) view;
            for (int i = 0; i < vg.getChildCount(); i++) {
                applyThemeToViewHierarchy(vg.getChildAt(i), color);
            }
        }
    }

    public static void applyColorToButton(Button button, int color) {
        if (button == null) return;
        Drawable bg = button.getBackground();
        if (bg != null) {
            bg.mutate().setColorFilter(color, PorterDuff.Mode.SRC_IN);
        } else {
            button.setBackgroundColor(color);
        }

        // Maintain high contrast text (white or black depending on color darkness)
        double darkness = 1 - (0.299 * Color.red(color) + 0.587 * Color.green(color) + 0.114 * Color.blue(color)) / 255;
        button.setTextColor(darkness > 0.4 ? Color.WHITE : Color.BLACK);
    }
}

    }

            return true;
        }

    }

import android.view.View;

public class ColorSatValView extends View {
    private Paint paint;
    private Paint thumbPaint;
    private Shader valShader;
    private float hue = 0f;
    private float sat = 0f;
    private float val = 1f;
    private OnColorChangedListener listener;

    public interface OnColorChangedListener {
        void onSatValChanged(float sat, float val);
    }

    public ColorSatValView(Context context, AttributeSet attrs) {
        super(context, attrs);
        paint = new Paint(Paint.ANTI_ALIAS_FLAG);
        thumbPaint = new Paint(Paint.ANTI_ALIAS_FLAG);
        thumbPaint.setStyle(Paint.Style.STROKE);
        thumbPaint.setStrokeWidth(5f);
    }

    public void setListener(OnColorChangedListener listener) {
        this.listener = listener;
    }

    public void setHue(float hue) {
        this.hue = hue;
        updateShader();
        invalidate();
    }

    public void setSatVal(float sat, float val) {
        this.sat = Math.max(0f, Math.min(1f, sat));
        this.val = Math.max(0f, Math.min(1f, val));
        invalidate();
    }

    @Override
    protected void onSizeChanged(int w, int h, int oldw, int oldh) {
        super.onSizeChanged(w, h, oldw, oldh);
        valShader = new LinearGradient(0, 0, 0, h, Color.TRANSPARENT, Color.BLACK, Shader.TileMode.CLAMP);
        updateShader();
    }

    private void updateShader() {
        int w = getWidth();
        if (w <= 0 || valShader == null) return;
        int pureHue = Color.HSVToColor(new float[]{hue, 1f, 1f});
        Shader satShader = new LinearGradient(0, 0, w, 0, Color.WHITE, pureHue, Shader.TileMode.CLAMP);
        ComposeShader composeShader = new ComposeShader(satShader, valShader, PorterDuff.Mode.SRC_OVER);
        paint.setShader(composeShader);
    }

    @Override
    protected void onDraw(Canvas canvas) {
        super.onDraw(canvas);
        int w = getWidth();
        int h = getHeight();
        canvas.drawRect(0, 0, w, h, paint);

        float cx = sat * w;
        float cy = (1f - val) * h;
        thumbPaint.setColor(val > 0.5f ? Color.BLACK : Color.WHITE);
        canvas.drawCircle(cx, cy, 14f, thumbPaint);
    }

    @Override
    public boolean onTouchEvent(MotionEvent event) {
        if (event.getAction() == MotionEvent.ACTION_DOWN || event.getAction() == MotionEvent.ACTION_MOVE) {
            float x = Math.max(0f, Math.min(getWidth(), event.getX()));
            float y = Math.max(0f, Math.min(getHeight(), event.getY()));
            sat = x / getWidth();
            val = 1f - (y / getHeight());
            invalidate();
            if (listener != null) listener.onSatValChanged(sat, val);
            return true;
        }
        return super.onTouchEvent(event);
    }
}

    public void onReceive(Context context, Intent intent) {
        super.onReceive(context, intent);
        String action = intent.getAction();
        Log.d(TAG, "onReceive action: " + action);

        if (action == null) return;

        AppWidgetManager appWidgetManager = AppWidgetManager.getInstance(context);
        int appWidgetId = intent.getIntExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, AppWidgetManager.INVALID_APPWIDGET_ID);

        // Uruchomienie goAsync() zapobiega uśpieniu procesu podczas zapytania do Azure
        final PendingResult pendingResult = goAsync();

        if (ACTION_PREV_DAY.equals(action) && appWidgetId != AppWidgetManager.INVALID_APPWIDGET_ID) {
            String currentDate = getSelectedDate(context, appWidgetId);
            String prevDate = MyDate.getPreviousDayDateSQLFriendlyFormat(currentDate);
            saveSelectedDate(context, appWidgetId, prevDate);
            fetchDataAndUpdateWidget(context, appWidgetManager, appWidgetId, pendingResult);

        } else if (ACTION_NEXT_DAY.equals(action) && appWidgetId != AppWidgetManager.INVALID_APPWIDGET_ID) {
            String currentDate = getSelectedDate(context, appWidgetId);
            String nextDate = MyDate.getNextDayDateSQLFriendlyFormat(currentDate);
            saveSelectedDate(context, appWidgetId, nextDate);
            fetchDataAndUpdateWidget(context, appWidgetManager, appWidgetId, pendingResult);

        } else if (ACTION_REFRESH.equals(action) || AppWidgetManager.ACTION_APPWIDGET_UPDATE.equals(action)) {
            ComponentName thisWidget = new ComponentName(context, DietWidgetProvider.class);
            int[] allWidgetIds = appWidgetManager.getAppWidgetIds(thisWidget);

            if (allWidgetIds.length == 0) {
                if (pendingResult != null) pendingResult.finish();
                return;
            }

            for (int i = 0; i < allWidgetIds.length; i++) {
                PendingResult pr = (i == allWidgetIds.length - 1) ? pendingResult : null;
                fetchDataAndUpdateWidget(context, appWidgetManager, allWidgetIds[i], pr);
            }
        } else {
            if (pendingResult != null) pendingResult.finish();
        }
    }

    @Override
    public void onDeleted(Context context, int[] appWidgetIds) {
        super.onDeleted(context, appWidgetIds);
        // Czyszczenie preferencji po usunięciu widżetu
        SharedPreferences.Editor editor = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE).edit();
        for (int widgetId : appWidgetIds) {
            editor.remove(PREF_PREFIX_KEY + widgetId);
        }
        editor.apply();
    }

    private static void fetchDataAndUpdateWidget(Context context, AppWidgetManager appWidgetManager, int appWidgetId, PendingResult pendingResult) {
        RemoteViews views = new RemoteViews(context.getPackageName(), R.layout.widget_calorie_tracker);

        int flags = PendingIntent.FLAG_UPDATE_CURRENT;
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            flags |= PendingIntent.FLAG_IMMUTABLE;
        }

        Intent settingsIntent = new Intent(context, WidgetColorSettingsActivity.class);
        PendingIntent settingsPendingIntent = PendingIntent.getActivity(context, 999, settingsIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetSettingsButton, settingsPendingIntent);

        // 1. Kliknięcie widżetu -> otwarcie DayData
        Intent openAppIntent = new Intent(context, DayData.class);
        PendingIntent openAppPendingIntent = PendingIntent.getActivity(context, 0, openAppIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetRoot, openAppPendingIntent);

        // 2. Kliknięcie: Poprzedni dzień (<)
        Intent prevIntent = new Intent(context, DietWidgetProvider.class);
        prevIntent.setAction(ACTION_PREV_DAY);
        prevIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        prevIntent.setData(Uri.parse("dietwidget://prev/" + appWidgetId));
        PendingIntent prevPendingIntent = PendingIntent.getBroadcast(context, appWidgetId * 10 + 1, prevIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetPrevDayButton, prevPendingIntent);

        // 3. Kliknięcie: Następny dzień (>)
        Intent nextIntent = new Intent(context, DietWidgetProvider.class);
        nextIntent.setAction(ACTION_NEXT_DAY);
        nextIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        nextIntent.setData(Uri.parse("dietwidget://next/" + appWidgetId));
        PendingIntent nextPendingIntent = PendingIntent.getBroadcast(context, appWidgetId * 10 + 2, nextIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetNextDayButton, nextPendingIntent);

        // 4. Kliknięcie: Odśwież (⟳)
        Intent refreshIntent = new Intent(context, DietWidgetProvider.class);
        refreshIntent.setAction(ACTION_REFRESH);
        refreshIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        refreshIntent.setData(Uri.parse("dietwidget://refresh/" + appWidgetId));
        PendingIntent refreshPendingIntent = PendingIntent.getBroadcast(context, appWidgetId * 10 + 3, refreshIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetRefreshButton, refreshPendingIntent);

        // 5. Podpięcie ListView
        Intent serviceIntent = new Intent(context, DietWidgetService.class);
        serviceIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        serviceIntent.setData(Uri.parse(serviceIntent.toUri(Intent.URI_INTENT_SCHEME)));
        views.setRemoteAdapter(R.id.widgetListView, serviceIntent);
        views.setEmptyView(R.id.widgetListView, R.id.widgetEmptyView);

        // 6. Ustalenie daty do wyświetlenia
        String selectedDate = getSelectedDate(context, appWidgetId);
        String todayDate = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).format(new Date());

        String displayDateText;
        if (selectedDate.equals(todayDate)) {
            displayDateText = "Today (" + selectedDate.substring(5) + ")";
        } else {
            displayDateText = selectedDate;
        }

        views.setTextViewText(R.id.widgetDateTextView, displayDateText);
        views.setTextViewText(R.id.widgetKcalTextView, "Pobieranie...");
        appWidgetManager.partiallyUpdateAppWidget(appWidgetId, views);

        // 7. Zapytanie SQL do Azure
        String query = QueryMaker.getAllProductsFromCalendarTableLikeDate(selectedDate);
        Log.d(TAG, "Fetching data for date: " + selectedDate + " | Query: " + query);

        SharedPreferences prefs = context.getSharedPreferences("DietWidgetPrefs", Context.MODE_PRIVATE);
        int bgColor = prefs.getInt("widget_bg_color", Color.WHITE);
        views.setInt(R.id.widgetRoot, "setBackgroundColor", bgColor);

// Jeśli tło jest ciemne, zmień kolor daty na jasny
        double darkness = 1 - (0.299 * Color.red(bgColor) + 0.587 * Color.green(bgColor) + 0.114 * Color.blue(bgColor)) / 255;
        int headerTextColor = (darkness > 0.5) ? Color.WHITE : Color.parseColor("#333333");
        views.setTextColor(R.id.widgetDateTextView, headerTextColor);

        GetFromSQLDatabase.runAzureQueryAIChat(query, new AzureDataCallback() {
            @Override
            public void onSuccess(ArrayList<RowInTable> resultTable) {
                float totalKcal = 0;
                float totalProtein = 0;
                float totalFat = 0;
                float totalCarbs = 0;

                ArrayList<RowInTable> rows = (resultTable != null) ? resultTable : new ArrayList<>();

                for (RowInTable row : rows) {
                    totalKcal += safeParse(row.getValue("kcal_consume"));
                    totalProtein += safeParse(row.getValue("protein_consume"));
                    totalFat += safeParse(row.getValue("fat_consume"));
                    totalCarbs += safeParse(row.getValue("carbs_consume"));
                }

                views.setTextViewText(R.id.widgetKcalTextView, String.format(Locale.getDefault(), "%.0f kcal", totalKcal));
                views.setTextViewText(R.id.widgetProteinTextView, String.format(Locale.getDefault(), "P: %.1fg", totalProtein));
                views.setTextViewText(R.id.widgetFatTextView, String.format(Locale.getDefault(), "F: %.1fg", totalFat));
                views.setTextViewText(R.id.widgetCarbsTextView, String.format(Locale.getDefault(), "C: %.1fg", totalCarbs));

                DietWidgetService.productList = rows;
                appWidgetManager.notifyAppWidgetViewDataChanged(appWidgetId, R.id.widgetListView);
                appWidgetManager.updateAppWidget(appWidgetId, views);

                if (pendingResult != null) pendingResult.finish();
            }

            @Override
            public void onFailure(String errorMessage) {
                Log.e(TAG, "Fetch failed: " + errorMessage);
                views.setTextViewText(R.id.widgetKcalTextView, "Błąd sieci");
                appWidgetManager.updateAppWidget(appWidgetId, views);

                if (pendingResult != null) pendingResult.finish();
            }
        });
    }

    private static float safeParse(String val) {
        if (val == null || val.trim().isEmpty() || val.equalsIgnoreCase("null")) return 0f;
        try {
            return Float.parseFloat(val);
        } catch (NumberFormatException e) {
            return 0f;
        }
    }

    public static void forceWidgetUpdate(Context context) {
        Intent intent = new Intent(context, DietWidgetProvider.class);
        intent.setAction(ACTION_REFRESH);
        context.sendBroadcast(intent);
    }
}

    public RemoteViewsFactory onGetViewFactory(Intent intent) {
        return new DietRemoteViewsFactory(this.getApplicationContext());

    static class DietRemoteViewsFactory implements RemoteViewsFactory {
        private final Context context;

        public DietRemoteViewsFactory(Context context) {
            this.context = context;
        }

        @Override
        public void onCreate() {}

        @Override
        public void onDataSetChanged() {
            // Wywoływane po notifyAppWidgetViewDataChanged
        }

        @Override
        public void onDestroy() {}

        @Override
        public int getCount() {
            return productList != null ? productList.size() : 0;
        }

        @Override
        public RemoteViews getViewAt(int position) {
            if (productList == null || position < 0 || position >= productList.size()) {
                return null;
            }

            RowInTable row = productList.get(position);
            RemoteViews views = new RemoteViews(context.getPackageName(), R.layout.widget_product_item);

            String name = row.getValue("product_name");
            String meal = row.getValue("meal_name");
            String amountStr = row.getValue("amount_of_product");
            String kcalStr = row.getValue("kcal_consume");

            float amount = 0f;
            float kcal = 0f;
            try {
                if (amountStr != null) amount = Float.parseFloat(amountStr);
                if (kcalStr != null) kcal = Float.parseFloat(kcalStr);
            } catch (NumberFormatException ignored) {}

            views.setTextViewText(R.id.itemProductName, name != null ? name : "Unknown");

            String details = (meal != null ? meal : "Meal") + " • " + String.format(Locale.getDefault(), "%.0fg", amount);
            views.setTextViewText(R.id.itemProductDetails, details);

            views.setTextViewText(R.id.itemProductKcal, String.format(Locale.getDefault(), "%.0f kcal", kcal));

            return views;
        }

        @Override
        public RemoteViews getLoadingView() {
            return null;
        }

        @Override
        public int getViewTypeCount() {
            return 1;
        }

        @Override
        public long getItemId(int position) {
            return position;
        }

            return true;
        }
    }

package com.lukaszjag.diet_tracker_android.tools.widget;

import android.content.Context;

import android.widget.Button;

import androidx.annotation.Nullable;
import androidx.appcompat.app.AppCompatActivity;

import com.lukaszjag.diet_tracker_android.R;

import java.util.Locale;

public class WidgetColorSettingsActivity extends AppCompatActivity {

    private ColorSatValView satValPicker;
    private ColorHueBarView huePicker;
    private ImageView colorPreview;
    private EditText etR, etG, etB, etH, etS, etV, etHex;
    private Button saveColorButton;

    private float currentHue = 0f;
    private float currentSat = 0f;
    private float currentVal = 1f;
    private boolean isUpdatingFromCode = false;

    @Override
    protected void onCreate(@Nullable Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_widget_color_settings);

        satValPicker = findViewById(R.id.satValPicker);
        huePicker = findViewById(R.id.huePicker);
        colorPreview = findViewById(R.id.colorPreview);

        etR = findViewById(R.id.etR);
        etG = findViewById(R.id.etG);
        etB = findViewById(R.id.etB);
        etH = findViewById(R.id.etH);
        etS = findViewById(R.id.etS);
        etV = findViewById(R.id.etV);
        etHex = findViewById(R.id.etHex);
        saveColorButton = findViewById(R.id.saveColorButton);

        // Odczyt wcześniej zapisanego koloru (domyślnie biały)
        SharedPreferences prefs = getSharedPreferences("DietWidgetPrefs", Context.MODE_PRIVATE);
        int savedColor = prefs.getInt("widget_bg_color", Color.WHITE);
        setColor(savedColor);

        // Listenery pickera
        satValPicker.setListener((sat, val) -> {
            currentSat = sat;
            currentVal = val;
            updateColorViews();
        });

        huePicker.setListener(hue -> {
            currentHue = hue;
            satValPicker.setHue(hue);
            updateColorViews();
        });

        // Wpisywanie ręczne HEX
        etHex.addTextChangedListener(new TextWatcher() {
            @Override public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            @Override public void onTextChanged(CharSequence s, int start, int before, int count) {
                if (isUpdatingFromCode) return;
                try {
                    String hex = s.toString().trim();
                    if (hex.length() == 6) {
                        int color = Color.parseColor("#" + hex);
                        setColor(color);
                    }
                } catch (Exception ignored) {}
            }
            @Override public void afterTextChanged(Editable s) {}
        });

        // Szablony
        findViewById(R.id.presetWhite).setOnClickListener(v -> setColor(0xFFFFFFFF));
        findViewById(R.id.presetDark).setOnClickListener(v -> setColor(0xFF252525));
        findViewById(R.id.presetAmoled).setOnClickListener(v -> setColor(0xFF000000));
        findViewById(R.id.presetBlue).setOnClickListener(v -> setColor(0xFF1E3A8A));

        // Zapis
        saveColorButton.setOnClickListener(v -> {
            int finalColor = Color.HSVToColor(new float[]{currentHue, currentSat, currentVal});
            prefs.edit().putInt("widget_bg_color", finalColor).apply();

            // Odśwież widżet na pulpicie
            DietWidgetProvider.forceWidgetUpdate(this);

            Toast.makeText(this, "Widget background updated!", Toast.LENGTH_SHORT).show();
            finish();
        });
    }

    private void setColor(int color) {
        float[] hsv = new float[3];
        Color.colorToHSV(color, hsv);
        currentHue = hsv[0];
        currentSat = hsv[1];
        currentVal = hsv[2];

        huePicker.setHue(currentHue);
        satValPicker.setHue(currentHue);
        satValPicker.setSatVal(currentSat, currentVal);

        updateColorViews();
    }

    private void updateColorViews() {
        int color = Color.HSVToColor(new float[]{currentHue, currentSat, currentVal});
        colorPreview.setBackgroundColor(color);

        isUpdatingFromCode = true;

        etR.setText(String.valueOf(Color.red(color)));
        etG.setText(String.valueOf(Color.green(color)));
        etB.setText(String.valueOf(Color.blue(color)));

        etH.setText(String.format(Locale.getDefault(), "%.0f", currentHue));
        etS.setText(String.format(Locale.getDefault(), "%.0f", currentSat * 100));
        etV.setText(String.format(Locale.getDefault(), "%.0f", currentVal * 100));

        String hex = String.format("%06X", (0xFFFFFF & color));
        etHex.setText(hex);

        isUpdatingFromCode = false;
    }
}
