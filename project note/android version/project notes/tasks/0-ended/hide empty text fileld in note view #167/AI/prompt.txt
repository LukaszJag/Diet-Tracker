    private void showCalendarDialog() {
        Map<String, Integer> countMap = new HashMap<>();
        for (Note note : adapter.getOriginalList()) {
            String created = note.getDateCreated();
            if (created != null && created.length() >= 10) {
                created = created.substring(0, 10);
                countMap.put(created, countMap.getOrDefault(created, 0) + 1);
            }
        }

        NotesCalendarDialog dialog = new NotesCalendarDialog(this, countMap, selectedDates, newSelectedDates -> {
            selectedDates.clear();
            selectedDates.addAll(newSelectedDates);
            updateDateFilterButtonUI();
            applyFilters();
        });
        dialog.show();
    }

    private void updateDateFilterButtonUI() {
        if (selectedDates.isEmpty()) {
            btnCalendarFilter.setText("📅 Choose Dates (All)");
            btnClearDateFilter.setVisibility(View.GONE);
        } else {
            btnCalendarFilter.setText("📅 Dates: " + selectedDates.size() + " selected");
            btnClearDateFilter.setVisibility(View.VISIBLE);
        }
    }

    private void fetchNotesFromAzure() {
        String query = QueryMaker.getAllNotesQuery();
        GetFromSQLDatabase.runAzureQueryAIChat(query, new AzureDataCallback() {
            @Override
            public void onSuccess(ArrayList<RowInTable> resultTable) {
                runOnUiThread(() -> {
                    adapter.clearAll();
                    for (RowInTable row : resultTable) {
                        String subcat = row.getValue("subcategory");
                        if (subcat == null) subcat = "";

                        boolean isDone = "true".equalsIgnoreCase(row.getValue("is_done")) || "1".equals(row.getValue("is_done"));
                        boolean isDeleted = "true".equalsIgnoreCase(row.getValue("is_deleted")) || "1".equals(row.getValue("is_deleted"));

                        Note note = new Note(
                                Long.parseLong(row.getValue("id")),
                                row.getValue("title"),
                                row.getValue("subtitle"),
                                row.getValue("description"),
                                row.getValue("category"),
                                subcat,
                                row.getValue("is_urgently"),
                                "true".equals(row.getValue("is_learning")) || "1".equals(row.getValue("is_learning")),
                                "true".equals(row.getValue("is_general_to_do")) || "1".equals(row.getValue("is_general_to_do")),
                                "true".equals(row.getValue("is_today_task")) || "1".equals(row.getValue("is_today_task")),
                                "true".equals(row.getValue("is_to_buy_task")) || "1".equals(row.getValue("is_to_buy_task")),
                                isDone,
                                isDeleted,
                                null,
                                row.getValue("date_created"),
                                row.getValue("date_deadline")
                        );
                        adapter.addItem(note);
                    }
                    applyFilters();
                });
            }

            @Override
            public void onFailure(String errorMessage) {
                runOnUiThread(() -> Toast.makeText(Notes.this, "Failed: " + errorMessage, Toast.LENGTH_SHORT).show());
            }
        });
    }

        boolean isEditMode = (note != null);

            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                String chosenMain = mainCategories.get(position);
                List<String> subcategories = NoteCategoryData.getSubcategories(chosenMain);
                List<String> displaySubs = new ArrayList<>();
                if (subcategories.isEmpty()) {
                    displaySubs.add("None");
                } else {
                    displaySubs.addAll(subcategories);
                }

                ArrayAdapter<String> subAdapter = new ArrayAdapter<>(Notes.this, android.R.layout.simple_spinner_item, displaySubs);
                subAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
                spinnerSubcategory.setAdapter(subAdapter);

                if (isEditMode && note.getNoteSubcategory() != null) {
                    int subIndex = displaySubs.indexOf(note.getNoteSubcategory());
                    if (subIndex != -1) {
                        spinnerSubcategory.setSelection(subIndex);
                    }
                }
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {}
        });

        if (isEditMode) {
            actionButtonsLayout.setVisibility(View.VISIBLE);

            title.setText(note.getNoteTitle());
            subtitle.setText(note.getNoteSubtitle());
            desc.setText(note.getNoteDescription());

            if (note.getNoteCategory() != null) {
                int catIndex = mainCategories.indexOf(note.getNoteCategory());
                if (catIndex != -1) {
                    spinnerCategory.setSelection(catIndex);
                }
            }

            cbLearn.setChecked(note.isLearning());
            cbGen.setChecked(note.isGeneralToDo());
            cbToday.setChecked(note.isTodayTask());
            cbBuy.setChecked(note.isToBuyTask());
        }

        // 1. Configure the dialog positive and negative buttons on builder FIRST
        builder.setPositiveButton(isEditMode ? "Update" : "Create", (d, which) -> {
            String selectedMain = spinnerCategory.getSelectedItem() != null ?
                    spinnerCategory.getSelectedItem().toString() : "";
            String selectedSub = spinnerSubcategory.getSelectedItem() != null ?
                    spinnerSubcategory.getSelectedItem().toString() : "";
            if ("None".equals(selectedSub)) {
                selectedSub = "";
            }

            String q;
            if (isEditMode) {
                q = QueryMaker.updateNoteQuery(
                        note.getId(),
                        title.getText().toString(),
                        subtitle.getText().toString(),
                        desc.getText().toString(),
                        selectedMain,
                        selectedSub,
                        false,
                        cbLearn.isChecked(),
                        cbGen.isChecked(),
                        cbToday.isChecked(),
                        cbBuy.isChecked(),
                        null
                );
            } else {
                String currentDate = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).format(new Date());
                q = QueryMaker.insertNoteQuery(
                        title.getText().toString(),
                        subtitle.getText().toString(),
                        desc.getText().toString(),
                        selectedMain,
                        selectedSub,
                        false,
                        cbLearn.isChecked(),
                        cbGen.isChecked(),
                        cbToday.isChecked(),
                        cbBuy.isChecked(),
                        currentDate,
                        null
                );
            }

            GetFromSQLDatabase.runAzureQueryAIChat(q, new AzureDataCallback() {
                @Override public void onSuccess(ArrayList<RowInTable> res) {
                    runOnUiThread(() -> {
                        Toast.makeText(Notes.this, isEditMode ? "Updated!" : "Saved!", Toast.LENGTH_SHORT).show();
                        fetchNotesFromAzure();
                    });
                }
                @Override public void onFailure(String err) {
                    runOnUiThread(() -> Toast.makeText(Notes.this, "Error: " + err, Toast.LENGTH_SHORT).show());
                }
            });
        });

        builder.setNegativeButton("Cancel", null);

        // 2. CREATE the dialog instance HERE (now it is initialized before being used in listeners)
        AlertDialog dialog = builder.create();

        // 3. Attach listeners to the custom view buttons (dialog.dismiss() works with no error)
        if (isEditMode) {
            btnMarkAsDone.setOnClickListener(v -> {
                String markDoneQuery = QueryMaker.markNoteAsDoneQuery(note.getId());
                GetFromSQLDatabase.runAzureQueryAIChat(markDoneQuery, new AzureDataCallback() {
                    @Override
                    public void onSuccess(ArrayList<RowInTable> resultTable) {
                        runOnUiThread(() -> {
                            Toast.makeText(Notes.this, "Marked as done!", Toast.LENGTH_SHORT).show();
                            dialog.dismiss();
                            fetchNotesFromAzure();
                        });
                    }

                    @Override
                    public void onFailure(String errorMessage) {
                        runOnUiThread(() -> Toast.makeText(Notes.this, "Error: " + errorMessage, Toast.LENGTH_SHORT).show());
                    }
                });
            });

            btnDelete.setOnClickListener(v -> {
                new AlertDialog.Builder(Notes.this)
                        .setTitle("Delete Note")
                        .setMessage("Move this note to deleted history?")
                        .setPositiveButton("Yes, Delete", (d, w) -> {
                            String deleteQuery = QueryMaker.markNoteAsDeletedQuery(note.getId());
                            GetFromSQLDatabase.runAzureQueryAIChat(deleteQuery, new AzureDataCallback() {
                                @Override
                                public void onSuccess(ArrayList<RowInTable> resultTable) {
                                    runOnUiThread(() -> {
                                        Toast.makeText(Notes.this, "Note deleted!", Toast.LENGTH_SHORT).show();
                                        dialog.dismiss();
                                        fetchNotesFromAzure();
                                    });
                                }

                                @Override
                                public void onFailure(String errorMessage) {
                                    runOnUiThread(() -> Toast.makeText(Notes.this, "Error: " + errorMessage, Toast.LENGTH_SHORT).show());
                                }
                            });
                        })
                        .setNegativeButton("Cancel", null)
                        .show();
            });
        }

        // 4. Finally show the dialog
        dialog.show();

        addListeners();
    }

    private void setupVisibilityListeners() {
        CheckBox cb1 = findViewById(R.id.hideCbSection1);
        CheckBox cb2 = findViewById(R.id.hideCbSection2);
        CheckBox cb3 = findViewById(R.id.hideCbSection3);
        CheckBox cb4 = findViewById(R.id.hideCbSection4);

        android.widget.CompoundButton.OnCheckedChangeListener listener = (buttonView, isChecked) -> {
            adapter.setSectionVisibilities(
                    cb1.isChecked(),
                    cb2.isChecked(),
                    cb3.isChecked(),
                    cb4.isChecked()
            );
        };

        cb1.setOnCheckedChangeListener(listener);
        cb2.setOnCheckedChangeListener(listener);
        cb3.setOnCheckedChangeListener(listener);
        cb4.setOnCheckedChangeListener(listener);
    }

    private void setupFilterListeners() {
        filterSubtitle.addTextChangedListener(new android.text.TextWatcher() {
            public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            public void onTextChanged(CharSequence s, int start, int before, int count) { applyFilters(); }
            public void afterTextChanged(android.text.Editable s) {}
        });

        filterCategorySpinner.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                String selectedMainCat = parent.getItemAtPosition(position).toString();
                updateSubcategoryFilterSpinner(selectedMainCat);
                applyFilters();
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {}
        });

        filterUrgentlySpinner.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                applyFilters();
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {}
        });

        android.widget.CompoundButton.OnCheckedChangeListener cbListener = (buttonView, isChecked) -> applyFilters();
        filterCbLearning.setOnCheckedChangeListener(cbListener);
        filterCbGeneral.setOnCheckedChangeListener(cbListener);
        filterCbToday.setOnCheckedChangeListener(cbListener);
        filterCbToBuy.setOnCheckedChangeListener(cbListener);
    }

    private void applyFilters() {
        String subtitle = filterSubtitle.getText().toString();

        String selectedCat = filterCategorySpinner.getSelectedItem() != null ?
                filterCategorySpinner.getSelectedItem().toString() : "All Categories";

        String selectedSubcat = filterUrgentlySpinner.getSelectedItem() != null ?
                filterUrgentlySpinner.getSelectedItem().toString() : "All Subcategories";

        // Filter through adapter
        adapter.filter(
                subtitle,
                selectedCat,
                selectedSubcat,
                filterCbLearning.isChecked(),
                filterCbGeneral.isChecked(),
                filterCbToday.isChecked(),
                filterCbToBuy.isChecked(),
                selectedDates
        );

        // Update the displayed notes counter
        updateNotesCountDisplay();
    }

    private void updateNotesCountDisplay() {
        if (tvNotesCount != null && adapter != null) {
            int displayedCount = adapter.getItemCount();
            int totalCount = adapter.getOriginalList().size();
            tvNotesCount.setText("Displayed: " + displayedCount + " (Total: " + totalCount + ")");
        }
    }

    private void addListeners(){
        pickDateDeadlineButton.setOnClickListener(v -> {
            MaterialDatePicker<Long> datePicker = MaterialDatePicker.Builder.datePicker()
                    .setTitleText("Select a Date")
                    .setSelection(MaterialDatePicker.todayInUtcMilliseconds())
                    .build();


            datePicker.addOnPositiveButtonClickListener(selection -> {
                // Ensure date formatting uses UTC to prevent 1-day timezone shifting
                SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault());
                sdf.setTimeZone(TimeZone.getTimeZone("UTC"));
                String formattedDate = sdf.format(new Date(selection));

                dialogDeadlineEditTextLocal.setText(formattedDate);
            });

            datePicker.show(getSupportFragmentManager(), "MATERIAL_DATE_PICKER");
        });
    }
}

    private void updateCalendar() {
        SimpleDateFormat sdf = new SimpleDateFormat("MMMM yyyy", Locale.getDefault());
        monthYearTextView.setText(sdf.format(currentDisplayCalendar.getTime()));

        // Calculate leading blanks (Monday-first: Mon=0 ... Sun=6)
        Calendar temp = (Calendar) currentDisplayCalendar.clone();
        temp.set(Calendar.DAY_OF_MONTH, 1);
        int firstDayOfWeek = temp.get(Calendar.DAY_OF_WEEK); // Sun=1, Mon=2
        int offset = (firstDayOfWeek == Calendar.SUNDAY) ? 6 : (firstDayOfWeek - Calendar.MONDAY);

        int daysInMonth = currentDisplayCalendar.getActualMaximum(Calendar.DAY_OF_MONTH);

        // Find max notes count in displayed month for proper gradient scaling
        int maxNotes = 1;
        int year = currentDisplayCalendar.get(Calendar.YEAR);
        int month = currentDisplayCalendar.get(Calendar.MONTH) + 1;
        for (int day = 1; day <= daysInMonth; day++) {
            String dateKey = String.format(Locale.getDefault(), "%04d-%02d-%02d", year, month, day);
            Integer count = notesCountPerDay.get(dateKey);
            if (count != null && count > maxNotes) {
                maxNotes = count;
            }
        }
        this.maxNotesCountInMonth = maxNotes;

        List<Integer> dayNumbers = new ArrayList<>();
        for (int i = 0; i < offset; i++) {
            dayNumbers.add(0); // empty cell placeholder
        }
        for (int day = 1; day <= daysInMonth; day++) {
            dayNumbers.add(day);
        }

        dayAdapter = new CalendarDayAdapter(dayNumbers);
        daysRecyclerView.setAdapter(dayAdapter);
        updateSelectedCountLabel();
    }

    private void updateSelectedCountLabel() {
        if (selectedDates.isEmpty()) {
            selectedCountTextView.setText("No days selected (Showing all notes)");
        } else {
            selectedCountTextView.setText("Selected: " + selectedDates.size() + " day(s)");
        }
    }

    // --- Grid Adapter for Days ---
    private class CalendarDayAdapter extends RecyclerView.Adapter<CalendarDayAdapter.DayViewHolder> {
        private final List<Integer> days;

        public CalendarDayAdapter(List<Integer> days) {
            this.days = days;
        }

        @NonNull
        @Override
        public DayViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
            View view = LayoutInflater.from(parent.getContext()).inflate(R.layout.item_calendar_day, parent, false);
            // Apply scale to cell height
            ViewGroup.LayoutParams lp = view.getLayoutParams();
            if (lp != null) {
                float density = getContext().getResources().getDisplayMetrics().density;
                lp.height = (int) (44 * density * uiScale);
                view.setLayoutParams(lp);
            }
            return new DayViewHolder(view);
        }

        @Override
        public void onBindViewHolder(@NonNull DayViewHolder holder, int position) {
            int dayNumber = days.get(position);
            if (dayNumber == 0) {
                holder.root.setVisibility(View.INVISIBLE);
                holder.root.setOnClickListener(null);
                return;
            }

            holder.root.setVisibility(View.VISIBLE);
            holder.tvDayNumber.setText(String.valueOf(dayNumber));
            holder.tvDayNumber.setTextSize(TypedValue.COMPLEX_UNIT_SP, 13 * uiScale);

            Calendar dayCal = (Calendar) currentDisplayCalendar.clone();
            dayCal.set(Calendar.DAY_OF_MONTH, dayNumber);
            boolean isSunday = (dayCal.get(Calendar.DAY_OF_WEEK) == Calendar.SUNDAY);

            // Sunday number MUST be dark red (#8B0000)
            if (isSunday) {
                holder.tvDayNumber.setTextColor(Color.parseColor("#8B0000"));
                holder.tvDayNumber.setTypeface(null, Typeface.BOLD);
            } else {
                holder.tvDayNumber.setTextColor(Color.parseColor("#222222"));
                holder.tvDayNumber.setTypeface(null, Typeface.NORMAL);
            }

            // Date string yyyy-MM-dd
            int year = currentDisplayCalendar.get(Calendar.YEAR);
            int month = currentDisplayCalendar.get(Calendar.MONTH) + 1;
            String dateKey = String.format(Locale.getDefault(), "%04d-%02d-%02d", year, month, dayNumber);

            // Note count heatmap calculation
            int noteCount = 0;
            if (notesCountPerDay.containsKey(dateKey)) {
                noteCount = notesCountPerDay.get(dateKey);
            }

            int backgroundColor;
            if (noteCount == 0) {
                // White if none
                backgroundColor = Color.WHITE;
            } else {
                // More intensive color as notes increase
                float intensity = 0.25f + 0.75f * ((float) noteCount / Math.max(1, maxNotesCountInMonth));
                backgroundColor = ColorUtils.blendARGB(Color.WHITE, heatmapBaseColor, Math.min(1.0f, intensity));
            }

            // Selection state
            boolean isSelected = selectedDates.contains(dateKey);
            GradientDrawable borderDrawable = new GradientDrawable();
            borderDrawable.setColor(backgroundColor);

            float density = getContext().getResources().getDisplayMetrics().density;
            if (isSelected) {
                borderDrawable.setStroke((int) (2.5f * density), Color.parseColor("#1565C0"));
                holder.ivCheck.setVisibility(View.VISIBLE);
            } else {
                borderDrawable.setStroke((int) (0.5f * density), Color.parseColor("#E0E0E0"));
                holder.ivCheck.setVisibility(View.GONE);
            }
            holder.root.setBackground(borderDrawable);

            // Toggle date selection on click
            holder.root.setOnClickListener(v -> {
                if (selectedDates.contains(dateKey)) {
                    selectedDates.remove(dateKey);
                } else {
                    selectedDates.add(dateKey);
                }
                notifyItemChanged(position);
                updateSelectedCountLabel();
            });
        }

        @Override
        public int getItemCount() {
            return days.size();
        }

        class DayViewHolder extends RecyclerView.ViewHolder {
            FrameLayout root;
            TextView tvDayNumber;
            TextView ivCheck;

            public DayViewHolder(@NonNull View itemView) {
                super(itemView);
                root = itemView.findViewById(R.id.calDayRoot);
                tvDayNumber = itemView.findViewById(R.id.calDayNumberTextView);
                ivCheck = itemView.findViewById(R.id.calDayCheckIndicator);
            }
        }
    }

public class MyAdapter extends RecyclerView.Adapter<MyAdapter.MyViewHolder> {

    public static final int SORT_NONE = 0;
    public static final int SORT_DATE_NEWEST = 1;
    public static final int SORT_DATE_OLDEST = 2;
    public static final int SORT_URGENCY_HIGH = 3;
    public static final int SORT_URGENCY_LOW = 4;

            section1 = itemView.findViewById(R.id.section1);

    private OnItemClickListener listener;

    private boolean hideSection2 = false;
    private boolean hideSection3 = false;
    private boolean hideSection4 = false;

        this.noteList = new ArrayList<>();

    public List<Note> getOriginalList() {
        return originalList;

        notifyDataSetChanged();
    }

    public void addItem(Note note) {
        noteList.add(note);
        originalList.add(note);
        applySort();
        notifyItemInserted(noteList.indexOf(note));
    }

    public void setItem(int position, Note note) {
        if (position >= 0 && position < noteList.size()) {
            Note oldNote = noteList.get(position);
            int origIdx = originalList.indexOf(oldNote);
            if (origIdx != -1) originalList.set(origIdx, note);
            noteList.set(position, note);
            applySort();
            notifyDataSetChanged();
        }
    }

    public void deleteItem(int position) {
        if (position >= 0 && position < noteList.size()) {
            Note note = noteList.get(position);
            originalList.remove(note);
            noteList.remove(position);
            notifyItemRemoved(position);
            notifyItemRangeChanged(position, noteList.size());
        }
    }

    private void applySort() {
        if (sortCriteria == SORT_NONE) return;
        Collections.sort(noteList, (n1, n2) -> {
            switch (sortCriteria) {
                case SORT_DATE_NEWEST:
                    return (n2.getDateCreated() != null ? n2.getDateCreated() : "")
                            .compareTo(n1.getDateCreated() != null ? n1.getDateCreated() : "");
                case SORT_DATE_OLDEST:
                    return (n1.getDateCreated() != null ? n1.getDateCreated() : "")
                            .compareTo(n2.getDateCreated() != null ? n2.getDateCreated() : "");
                case SORT_URGENCY_HIGH:
                    return Integer.compare(getUrgencyWeight(n2), getUrgencyWeight(n1));
                case SORT_URGENCY_LOW:
                    return Integer.compare(getUrgencyWeight(n1), getUrgencyWeight(n2));
                default:
                    return 0;
            }
        });
    }

    private int getUrgencyWeight(Note note) {
        if (note == null) return 0;
        int idx = note.getUrgentScaleEnglish().indexOf(note.getNoteUrgently());
        return (idx != -1) ? idx : 0;
    }

        filter(subtitle, selectedMainCategory, selectedSubcategory, showLearningOnly, showGeneralOnly, showTodayOnly, showToBuyOnly, statusFilter, null);
    }

    public void filter(String subtitle, String selectedMainCategory, String selectedSubcategory,
                       boolean showLearningOnly, boolean showGeneralOnly, boolean showTodayOnly, boolean showToBuyOnly,
                       String statusFilter, Set<String> selectedDates) {
        noteList.clear();
        String qSub = (subtitle != null) ? subtitle.toLowerCase().trim() : "";
        boolean isAllCategories = (selectedMainCategory == null || selectedMainCategory.equals("All Categories"));
        boolean isAllSubcategories = (selectedSubcategory == null || selectedSubcategory.equals("All Subcategories"));

        boolean hasDateFilter = (selectedDates != null && !selectedDates.isEmpty());

        for (Note note : originalList) {
            boolean matchSub = qSub.isEmpty() || (note.getNoteSubtitle() != null && note.getNoteSubtitle().toLowerCase().contains(qSub));

            boolean matchCat = true;
            if (!isAllCategories) {
                matchCat = note.getNoteCategory() != null && note.getNoteCategory().equalsIgnoreCase(selectedMainCategory.trim());
                if (matchCat && !isAllSubcategories) {
                    matchCat = note.getNoteSubcategory() != null && note.getNoteSubcategory().equalsIgnoreCase(selectedSubcategory.trim());
                }
            }

            boolean matchLearning = !showLearningOnly || note.isLearning();
            boolean matchGeneral = !showGeneralOnly || note.isGeneralToDo();
            boolean matchToday = !showTodayOnly || note.isTodayTask();
            boolean matchToBuy = !showToBuyOnly || note.isToBuyTask();

            boolean matchStatus = true;
            if ("DONE".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDone();
            } else if ("DELETED".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDeleted();
            }

            boolean matchDate = true;
            if (hasDateFilter) {
                String created = note.getDateCreated();
                if (created != null && created.length() >= 10) {
                    created = created.substring(0, 10);
                }
                matchDate = (created != null && selectedDates.contains(created));
            }

            if (matchSub && matchCat && matchLearning && matchGeneral && matchToday && matchToBuy && matchStatus && matchDate) {
                noteList.add(note);
            }
        }
        applySort();
        notifyDataSetChanged();
    }

        holder.section4.setVisibility(hideSection4 ? View.GONE : View.VISIBLE);

        String titlePrefix = "";
        holder.tv1.setPaintFlags(holder.tv1.getPaintFlags() & (~Paint.STRIKE_THRU_TEXT_FLAG));
        holder.tv1.setTextColor(Color.BLACK);

        if (currentItem.isDeleted()) {
            titlePrefix = "[DELETED] ";
            holder.tv1.setTextColor(Color.RED);
            holder.tv1.setPaintFlags(holder.tv1.getPaintFlags() | Paint.STRIKE_THRU_TEXT_FLAG);
        } else if (currentItem.isDone()) {
            titlePrefix = "[DONE ✓] ";
            holder.tv1.setTextColor(Color.parseColor("#2E7D32"));
        }

        holder.tv1.setText(titlePrefix + currentItem.getNoteTitle());
        holder.tv2.setText(currentItem.getNoteSubtitle());
        holder.tv3.setText(currentItem.getNoteDescription());

        String catDisplay = (currentItem.getNoteCategory() != null && !currentItem.getNoteCategory().isEmpty())
                ? currentItem.getNoteCategory() : "-";
        if (currentItem.getNoteSubcategory() != null && !currentItem.getNoteSubcategory().trim().isEmpty()) {
            catDisplay += " / " + currentItem.getNoteSubcategory();
        }
        holder.tvCategory.setText("Category: " + catDisplay);

        holder.tvUrgently.setText("Urgently: " + (currentItem.getNoteUrgently() != null ? currentItem.getNoteUrgently() : "-"));

        holder.cbIsLearning.setChecked(currentItem.isLearning());
        holder.cbIsGeneralToDo.setChecked(currentItem.isGeneralToDo());
        holder.cbIsTodayTask.setChecked(currentItem.isTodayTask());
        holder.cbIsToBuy.setChecked(currentItem.isToBuyTask());

        String createdDate = currentItem.getDateCreated() != null ? currentItem.getDateCreated() : "-";
        String deadlineDate = (currentItem.getDateDeadline() != null && !currentItem.getDateDeadline().isEmpty())
                ? currentItem.getDateDeadline() : "None";
        holder.tvDates.setText("Created: " + createdDate + " | Deadline: " + deadlineDate);
        holder.tvDaysSince.setText(currentItem.getDaysSinceCreation() + " days since create");

            int pos = holder.getAdapterPosition();
            if (listener != null && pos != RecyclerView.NO_POSITION) listener.onItemClick(pos, noteList.get(pos));
        });
    }

package com.lukaszjag.diet_tracker_android.tools.notes_tool;

import android.content.Context;
import android.net.Uri;
import android.os.Environment;

import org.json.JSONArray;
import org.json.JSONObject;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;

public class NoteStorage {
    private static final String FILE_NAME = "notes_data.json";

    // Resolves dynamically to the standard Download/diet-tracker-data directory on any Android device
    private static final File TARGET_DIRECTORY = new File(
            Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS),
            "diet-tracker-data"
    );

    /**
     * Resolves the target directory and ensures the folder structure is created.
     */
    private static File getTargetFile() {
        if (!TARGET_DIRECTORY.exists()) {
            TARGET_DIRECTORY.mkdirs(); // Automatically creates "diet-tracker-data" if missing
        }
        return new File(TARGET_DIRECTORY, FILE_NAME);
    }

    /**
     * Saves the list of notes to the specified external storage path.
     */
    public static void saveNotes(Context context, List<Note> notes) {
        try {
            JSONArray jsonArray = new JSONArray();
            for (Note note : notes) {
                JSONObject noteJson = note.toJsonObject();
                if (noteJson != null) {
                    jsonArray.put(noteJson);
                }
            }
            String jsonContent = jsonArray.toString(4);

            File file = getTargetFile();
            try (FileOutputStream fos = new FileOutputStream(file);
                 OutputStreamWriter osw = new OutputStreamWriter(fos, StandardCharsets.UTF_8)) {
                osw.write(jsonContent);
            }
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error saving notes", e);
        }
    }

    /**
     * Loads the list of notes from the specified external storage path.
     */
    public static List<Note> loadNotes(Context context) {
        List<Note> notes = new ArrayList<>();
        File file = getTargetFile();

        if (!file.exists()) {
            return notes;
        }

        try (FileInputStream fis = new FileInputStream(file);
             InputStreamReader isr = new InputStreamReader(fis, StandardCharsets.UTF_8);
             BufferedReader reader = new BufferedReader(isr)) {

            StringBuilder sb = new StringBuilder();
            char[] buffer = new char[1024];
            int numRead;
            while ((numRead = reader.read(buffer)) != -1) {
                sb.append(buffer, 0, numRead);
            }

            String jsonContent = sb.toString();

            // Strip hidden UTF-8 BOM if present (often added by Windows/external text editors)
            if (jsonContent.startsWith("\uFEFF")) {
                jsonContent = jsonContent.substring(1);
            }
            jsonContent = jsonContent.trim();

            if (!jsonContent.isEmpty()) {
                JSONArray jsonArray = new JSONArray(jsonContent);
                for (int i = 0; i < jsonArray.length(); i++) {
                    JSONObject noteJson = jsonArray.getJSONObject(i);
                    Note note = Note.fromJsonObject(noteJson);
                    if (note != null) {
                        notes.add(note);
                    }
                }
            }
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error reading or parsing notes JSON file", e);
        }

        return notes;
    }

    /**
     * Imports external JSON files via Uri, validates structure, and overwrites local notes storage.
     */
    public static boolean importNotes(Context context, Uri sourceUri) {
        try (InputStream in = context.getContentResolver().openInputStream(sourceUri)) {
            if (in == null) return false;

            BufferedReader reader = new BufferedReader(new InputStreamReader(in, StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            char[] buffer = new char[1024];
            int numRead;
            while ((numRead = reader.read(buffer)) != -1) {
                sb.append(buffer, 0, numRead);
            }

            String jsonContent = sb.toString();

            // Strip hidden UTF-8 BOM if present
            if (jsonContent.startsWith("\uFEFF")) {
                jsonContent = jsonContent.substring(1);
            }
            jsonContent = jsonContent.trim();

            JSONArray jsonArray = new JSONArray(jsonContent);
            List<Note> importedNotes = new ArrayList<>();
            for (int i = 0; i < jsonArray.length(); i++) {
                Note note = Note.fromJsonObject(jsonArray.getJSONObject(i));
                if (note != null) {
                    importedNotes.add(note);
                }
            }

            // Write validated imported notes directly to the target external storage
            saveNotes(context, importedNotes);
            return true;
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error importing notes", e);
            return false;
        }
    }
}
