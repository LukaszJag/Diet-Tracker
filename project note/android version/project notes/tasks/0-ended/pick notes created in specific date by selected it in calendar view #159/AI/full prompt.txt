package com.lukaszjag.diet_tracker_android;

import android.content.Intent;
import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;
import android.view.View;
import android.widget.Button;

import androidx.appcompat.app.AppCompatActivity;

import com.lukaszjag.diet_tracker_android.gui.diet.GetProductData;
import com.lukaszjag.diet_tracker_android.gui.diet.day_data_view.DayData;
import com.lukaszjag.diet_tracker_android.gui.notes.Notes;
import com.lukaszjag.diet_tracker_android.gui.notes.NotesHistory;

public class MainActivity extends AppCompatActivity {

    private Button addMealButton;
    private Button getDayDataButton;
    private Button noteTasksButton;
    private Button notesHistoryButton;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.main_window);

        setupUIComponents();
        addListeners();
    }

    private void setupUIComponents() {
        addMealButton = findViewById(R.id.addMealButton);
        getDayDataButton = findViewById(R.id.getDayDataButton);
        noteTasksButton = findViewById(R.id.noteTaskButtonButton);
        notesHistoryButton = findViewById(R.id.notesHistoryButton);
    }

    private void addListeners() {
        noteTasksButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, Notes.class);
            startActivity(intent);
        });

        notesHistoryButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, NotesHistory.class);
            startActivity(intent);
        });

        addMealButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, GetProductData.class);
            startActivity(intent);
        });

        getDayDataButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, DayData.class);
            startActivity(intent);
        });
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        getMenuInflater().inflate(R.menu.menu_main, menu);
        return true;
    }

    @Override
    public boolean onOptionsItemSelected(MenuItem item) {
        if (item.getItemId() == R.id.action_settings) {
            return true;
        }
        return super.onOptionsItemSelected(item);
    }
}

    public static String[] getAllDaysInMonthAndYearSQLFriendlyFormat(int month, int year) {
        String[] allDaysInSQLFormat = new String[getAmountOfDaysInMonth(month)];

        String monthPart = "";
        if (month < 10) {
            monthPart = "0" + month;
        } else {
            monthPart = "" + month;
        }

        for (int i = 0; i < allDaysInSQLFormat.length; i++) {
            if (i < 10) {
                allDaysInSQLFormat[i] = "" + year + "-" + monthPart + "-" + "0" + (i + 1);
            } else {
                allDaysInSQLFormat[i] = "" + year + "-" + monthPart + "-" + (i + 1);
            }
        }

        return allDaysInSQLFormat;
    }

public class Notes extends AppCompatActivity {

    }

    private void setupFilterSpinners() {
        List<String> mainCategoriesFilter = new ArrayList<>();
        mainCategoriesFilter.add("All Categories");
        mainCategoriesFilter.addAll(NoteCategoryData.getMainCategories());

        ArrayAdapter<String> mainCatAdapter = new ArrayAdapter<>(this, android.R.layout.simple_spinner_item, mainCategoriesFilter);
        mainCatAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        filterCategorySpinner.setAdapter(mainCatAdapter);

        updateSubcategoryFilterSpinner("All Categories");
    }

    private void updateSubcategoryFilterSpinner(String mainCategory) {
        List<String> subcategoriesFilter = new ArrayList<>();
        subcategoriesFilter.add("All Subcategories");

        if (mainCategory != null && !mainCategory.equals("All Categories")) {
            List<String> subs = NoteCategoryData.getSubcategories(mainCategory);
            if (subs != null && !subs.isEmpty()) {
                subcategoriesFilter.addAll(subs);
            }
        }

        ArrayAdapter<String> subcatAdapter = new ArrayAdapter<>(this, android.R.layout.simple_spinner_item, subcategoriesFilter);
        subcatAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        filterUrgentlySpinner.setAdapter(subcatAdapter);
    }

    private void fetchNotesFromAzure() {
        // QueryMaker.getAllNotesQuery() selects only active items
        String query = QueryMaker.getAllNotesQuery();
        GetFromSQLDatabase.runAzureQueryAIChat(query, new AzureDataCallback() {
            @Override
            public void onSuccess(ArrayList<RowInTable> resultTable) {
                runOnUiThread(() -> {
                    adapter.clearAll();
                    for (RowInTable row : resultTable) {
                        String subcat = row.getValue("subcategory");
                        if (subcat == null) subcat = "";

                        boolean isDone = "true".equalsIgnoreCase(row.getValue("is_done")) || "1".equals(row.getValue("is_done"));
                        boolean isDeleted = "true".equalsIgnoreCase(row.getValue("is_deleted")) || "1".equals(row.getValue("is_deleted"));

                        Note note = new Note(
                                Long.parseLong(row.getValue("id")),
                                row.getValue("title"),
                                row.getValue("subtitle"),
                                row.getValue("description"),
                                row.getValue("category"),
                                subcat,
                                row.getValue("is_urgently"),
                                "true".equals(row.getValue("is_learning")) || "1".equals(row.getValue("is_learning")),
                                "true".equals(row.getValue("is_general_to_do")) || "1".equals(row.getValue("is_general_to_do")),
                                "true".equals(row.getValue("is_today_task")) || "1".equals(row.getValue("is_today_task")),
                                "true".equals(row.getValue("is_to_buy_task")) || "1".equals(row.getValue("is_to_buy_task")),
                                isDone,
                                isDeleted,
                                null,
                                row.getValue("date_created"),
                                row.getValue("date_deadline")
                        );
                        adapter.addItem(note);
                    }
                    applyFilters();
                });
            }

            @Override
            public void onFailure(String errorMessage) {
                runOnUiThread(() -> Toast.makeText(Notes.this, "Failed: " + errorMessage, Toast.LENGTH_SHORT).show());
            }
        });
    }

    private void showNoteDialog(Note note) {
        AlertDialog.Builder builder = new AlertDialog.Builder(this);
        View dialogView = getLayoutInflater().inflate(R.layout.dialog_note, null);
        builder.setView(dialogView);

        final EditText title = dialogView.findViewById(R.id.dialogTitle);
        final EditText subtitle = dialogView.findViewById(R.id.dialogSubtitle);
        final EditText desc = dialogView.findViewById(R.id.dialogDescription);
        final Spinner spinnerCategory = dialogView.findViewById(R.id.dialogSpinnerCategory);
        final Spinner spinnerSubcategory = dialogView.findViewById(R.id.dialogSpinnerSubcategory);
        final CheckBox cbLearn = dialogView.findViewById(R.id.dialogCbLearning);
        final CheckBox cbGen = dialogView.findViewById(R.id.dialogCbGeneral);
        final CheckBox cbToday = dialogView.findViewById(R.id.dialogCbToday);
        final CheckBox cbBuy = dialogView.findViewById(R.id.dialogCbToBuy);

        final LinearLayout actionButtonsLayout = dialogView.findViewById(R.id.dialogActionButtonsLayout);
        final Button btnMarkAsDone = dialogView.findViewById(R.id.dialogBtnMarkAsDone);
        final Button btnDelete = dialogView.findViewById(R.id.dialogBtnDelete);

        boolean isEditMode = (note != null);

        List<String> mainCategories = NoteCategoryData.getMainCategories();
        ArrayAdapter<String> catAdapter = new ArrayAdapter<>(this, android.R.layout.simple_spinner_item, mainCategories);
        catAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        spinnerCategory.setAdapter(catAdapter);

        spinnerCategory.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                String chosenMain = mainCategories.get(position);
                List<String> subcategories = NoteCategoryData.getSubcategories(chosenMain);
                List<String> displaySubs = new ArrayList<>();
                if (subcategories.isEmpty()) {
                    displaySubs.add("None");
                } else {
                    displaySubs.addAll(subcategories);
                }

                ArrayAdapter<String> subAdapter = new ArrayAdapter<>(Notes.this, android.R.layout.simple_spinner_item, displaySubs);
                subAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
                spinnerSubcategory.setAdapter(subAdapter);

                if (isEditMode && note.getNoteSubcategory() != null) {
                    int subIndex = displaySubs.indexOf(note.getNoteSubcategory());
                    if (subIndex != -1) {
                        spinnerSubcategory.setSelection(subIndex);
                    }
                }
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {}
        });

        if (isEditMode) {
            actionButtonsLayout.setVisibility(View.VISIBLE);

            title.setText(note.getNoteTitle());
            subtitle.setText(note.getNoteSubtitle());
            desc.setText(note.getNoteDescription());

            if (note.getNoteCategory() != null) {
                int catIndex = mainCategories.indexOf(note.getNoteCategory());
                if (catIndex != -1) {
                    spinnerCategory.setSelection(catIndex);
                }
            }

            cbLearn.setChecked(note.isLearning());
            cbGen.setChecked(note.isGeneralToDo());
            cbToday.setChecked(note.isTodayTask());
            cbBuy.setChecked(note.isToBuyTask());
        }

        AlertDialog dialog = builder.create();

        if (isEditMode) {
            btnMarkAsDone.setOnClickListener(v -> {
                String markDoneQuery = QueryMaker.markNoteAsDoneQuery(note.getId());
                GetFromSQLDatabase.runAzureQueryAIChat(markDoneQuery, new AzureDataCallback() {
                    @Override
                    public void onSuccess(ArrayList<RowInTable> resultTable) {
                        runOnUiThread(() -> {
                            Toast.makeText(Notes.this, "Marked as done!", Toast.LENGTH_SHORT).show();
                            dialog.dismiss();
                            fetchNotesFromAzure();
                        });
                    }

                    @Override
                    public void onFailure(String errorMessage) {
                        runOnUiThread(() -> Toast.makeText(Notes.this, "Error: " + errorMessage, Toast.LENGTH_SHORT).show());
                    }
                });
            });

            btnDelete.setOnClickListener(v -> {
                new AlertDialog.Builder(Notes.this)
                        .setTitle("Delete Note")
                        .setMessage("Move this note to deleted history?")
                        .setPositiveButton("Yes, Delete", (d, w) -> {
                            String deleteQuery = QueryMaker.markNoteAsDeletedQuery(note.getId());
                            GetFromSQLDatabase.runAzureQueryAIChat(deleteQuery, new AzureDataCallback() {
                                @Override
                                public void onSuccess(ArrayList<RowInTable> resultTable) {
                                    runOnUiThread(() -> {
                                        Toast.makeText(Notes.this, "Note deleted!", Toast.LENGTH_SHORT).show();
                                        dialog.dismiss();
                                        fetchNotesFromAzure();
                                    });
                                }

                                @Override
                                public void onFailure(String errorMessage) {
                                    runOnUiThread(() -> Toast.makeText(Notes.this, "Error: " + errorMessage, Toast.LENGTH_SHORT).show());
                                }
                            });
                        })
                        .setNegativeButton("Cancel", null)
                        .show();
            });
        }

        builder.setPositiveButton(isEditMode ? "Update" : "Create", (d, which) -> {
            String selectedMain = spinnerCategory.getSelectedItem() != null ?
                    spinnerCategory.getSelectedItem().toString() : "";
            String selectedSub = spinnerSubcategory.getSelectedItem() != null ?
                    spinnerSubcategory.getSelectedItem().toString() : "";
            if ("None".equals(selectedSub)) {
                selectedSub = "";
            }

            String q;
            if (isEditMode) {
                q = QueryMaker.updateNoteQuery(
                        note.getId(),
                        title.getText().toString(),
                        subtitle.getText().toString(),
                        desc.getText().toString(),
                        selectedMain,
                        selectedSub,
                        false,
                        cbLearn.isChecked(),
                        cbGen.isChecked(),
                        cbToday.isChecked(),
                        cbBuy.isChecked(),
                        null
                );
            } else {
                String currentDate = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).format(new Date());
                q = QueryMaker.insertNoteQuery(
                        title.getText().toString(),
                        subtitle.getText().toString(),
                        desc.getText().toString(),
                        selectedMain,
                        selectedSub,
                        false,
                        cbLearn.isChecked(),
                        cbGen.isChecked(),
                        cbToday.isChecked(),
                        cbBuy.isChecked(),
                        currentDate,
                        null
                );
            }

            GetFromSQLDatabase.runAzureQueryAIChat(q, new AzureDataCallback() {
                @Override public void onSuccess(ArrayList<RowInTable> res) {
                    runOnUiThread(() -> {
                        Toast.makeText(Notes.this, isEditMode ? "Updated!" : "Saved!", Toast.LENGTH_SHORT).show();
                        fetchNotesFromAzure();
                    });
                }
                @Override public void onFailure(String err) {
                    runOnUiThread(() -> Toast.makeText(Notes.this, "Error: " + err, Toast.LENGTH_SHORT).show());
                }
            });
        });

        builder.setNegativeButton("Cancel", null);
        builder.show();
    }

    private void setupFilterListeners() {

import android.util.Log;

public class GetFromSQLDatabase {

    public void runCustomAzureQuery(String mySqlString) {
        AzureApiService apiService = RetrofitClient.getRetrofitInstance().create(AzureApiService.class);
        QueryRequest requestBody = new QueryRequest(mySqlString);
        Call<List<Map<String, Object>>> call = apiService.executeCustomQuery(requestBody);

        call.enqueue(new Callback<List<Map<String, Object>>>() {
            @Override
            public void onResponse(Call<List<Map<String, Object>>> call, Response<List<Map<String, Object>>> response) {
                if (response.isSuccessful() && response.body() != null) {
                    List<Map<String, Object>> dynamicSqlData = response.body();
                    if (!dynamicSqlData.isEmpty()) {
                        Log.d("AZURE_CUSTOM_SQL", "amount of rows: " + dynamicSqlData.size());
                        for (int i = 0; i < dynamicSqlData.size(); i++) {
                            Map<String, Object> row = dynamicSqlData.get(i);
                            Log.d("AZURE_CUSTOM_SQL", "Row data: [" + i + "]: " + row.get("product_name"));
                        }
                    } else {
                        Log.d("AZURE_DATABASE_SQL", "No result");
                    }
                } else {
                    Log.d("AZURE_ERROR", "response.isSuccessful() or response.body() != null");
                }
            }

            @Override
            public void onFailure(Call<List<Map<String, Object>>> call, Throwable t) {
                Log.e("AZURE_CUSTOM_SQL", "Network fail: " + t.getMessage());
            }
        });
    }

    public static void runAzureQueryAIChat(String mySqlString, AzureDataCallback callback) {
        AzureApiService apiService = RetrofitClient.getRetrofitInstance().create(AzureApiService.class);
        QueryRequest requestBody = new QueryRequest(mySqlString);

        Log.d("AZURE_DEBUG", "Sending Query: " + mySqlString);

        Call<List<Map<String, Object>>> call = apiService.executeCustomQuery(requestBody);

        call.enqueue(new Callback<List<Map<String, Object>>>() {
            @Override
            public void onResponse(Call<List<Map<String, Object>>> call, Response<List<Map<String, Object>>> response) {
                ArrayList<RowInTable> resultTable = new ArrayList<>();

                Log.d("AZURE_DEBUG", "Response Code: " + response.code());

                if (response.isSuccessful()) {
                    List<Map<String, Object>> dynamicSqlData = response.body();

                    if (dynamicSqlData != null && !dynamicSqlData.isEmpty()) {
                        for (int i = 0; i < dynamicSqlData.size(); i++) {
                            Map<String, Object> row = dynamicSqlData.get(i);
                            RowInTable rowInTable = new RowInTable();

                            for (String key : row.keySet()) {
                                rowInTable.putKeyAndValueToRow(key, String.valueOf(row.get(key)));
                            }
                            resultTable.add(rowInTable);
                        }
                        callback.onSuccess(resultTable);
                    } else {
                        Log.d("AZURE_DEBUG", "Query executed successfully with empty result set.");
                        callback.onSuccess(new ArrayList<>());
                    }
                } else {
                    String detailedError = "Unknown error";
                    try {
                        if (response.errorBody() != null) {
                            detailedError = response.errorBody().string();
                        }
                    } catch (IOException e) {
                        Log.e("AZURE_DEBUG", "Failed to parse error body: " + e.getMessage());
                    }

                    Log.e("AZURE_ERROR_DETAILS", "HTTP Error " + response.code() + ": " + detailedError);
                    callback.onFailure("Server error (" + response.code() + "): " + detailedError);
                }
            }

            @Override
            public void onFailure(Call<List<Map<String, Object>>> call, Throwable t) {
                Log.e("AZURE_DEBUG", "Network Transport Failure: " + t.getMessage(), t);
                callback.onFailure("Network fail: " + t.getMessage());
            }
        });
    }

package com.lukaszjag.diet_tracker_android.tools.cloud_data_tools;

import java.io.IOException;
import java.lang.annotation.Annotation;
import java.lang.reflect.Type;
import java.util.ArrayList;

import okhttp3.ResponseBody;
import retrofit2.Converter;
import retrofit2.Retrofit;

public class NullOnEmptyConverterFactory extends Converter.Factory {
    @Override
    public Converter<ResponseBody, ?> responseBodyConverter(Type type, Annotation[] annotations, Retrofit retrofit) {
        final Converter<ResponseBody, ?> delegate = retrofit.nextResponseBodyConverter(this, type, annotations);
        return new Converter<ResponseBody, Object>() {
            @Override
            public Object convert(ResponseBody body) throws IOException {
                if (body.contentLength() == 0) {
                    return new ArrayList<>();
                }
                String content = body.string();
                if (content == null || content.trim().isEmpty() || content.trim().equalsIgnoreCase("null")) {
                    return new ArrayList<>();
                }
                // Reconstruct the response body with the read content for Gson
                ResponseBody newBody = ResponseBody.create(body.contentType(), content);
                return delegate.convert(newBody);
            }
        };
    }
}

    private void applySort() {
        if (sortCriteria == SORT_NONE) return;
        Collections.sort(noteList, (n1, n2) -> {
            switch (sortCriteria) {
                case SORT_DATE_NEWEST:
                    return (n2.getDateCreated() != null ? n2.getDateCreated() : "")
                            .compareTo(n1.getDateCreated() != null ? n1.getDateCreated() : "");
                case SORT_DATE_OLDEST:
                    return (n1.getDateCreated() != null ? n1.getDateCreated() : "")
                            .compareTo(n2.getDateCreated() != null ? n2.getDateCreated() : "");
                case SORT_URGENCY_HIGH:
                    return Integer.compare(getUrgencyWeight(n2), getUrgencyWeight(n1));
                case SORT_URGENCY_LOW:
                    return Integer.compare(getUrgencyWeight(n1), getUrgencyWeight(n2));
                default:
                    return 0;
            }
        });
    }

    private int getUrgencyWeight(Note note) {
        if (note == null) return 0;
        int idx = note.getUrgentScaleEnglish().indexOf(note.getNoteUrgently());
        return (idx != -1) ? idx : 0;
    }

    public void filter(String subtitle, String selectedMainCategory, String selectedSubcategory,
                       boolean showLearningOnly, boolean showGeneralOnly, boolean showTodayOnly, boolean showToBuyOnly,
                       String statusFilter) {
        noteList.clear();
        String qSub = (subtitle != null) ? subtitle.toLowerCase().trim() : "";
        boolean isAllCategories = (selectedMainCategory == null || selectedMainCategory.equals("All Categories"));
        boolean isAllSubcategories = (selectedSubcategory == null || selectedSubcategory.equals("All Subcategories"));

        for (Note note : originalList) {
            boolean matchSub = qSub.isEmpty() || (note.getNoteSubtitle() != null && note.getNoteSubtitle().toLowerCase().contains(qSub));

            boolean matchCat = true;
            if (!isAllCategories) {
                matchCat = note.getNoteCategory() != null && note.getNoteCategory().equalsIgnoreCase(selectedMainCategory.trim());
                if (matchCat && !isAllSubcategories) {
                    matchCat = note.getNoteSubcategory() != null && note.getNoteSubcategory().equalsIgnoreCase(selectedSubcategory.trim());
                }
            }

            boolean matchLearning = !showLearningOnly || note.isLearning();
            boolean matchGeneral = !showGeneralOnly || note.isGeneralToDo();
            boolean matchToday = !showTodayOnly || note.isTodayTask();
            boolean matchToBuy = !showToBuyOnly || note.isToBuyTask();

            // Status filter for history
            boolean matchStatus = true;
            if ("DONE".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDone();
            } else if ("DELETED".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDeleted();
            }

            if (matchSub && matchCat && matchLearning && matchGeneral && matchToday && matchToBuy && matchStatus) {
                noteList.add(note);
            }
        }
        applySort();
        notifyDataSetChanged();
    }

import android.content.Context;
import android.net.Uri;
import android.os.Environment;

import org.json.JSONArray;
import org.json.JSONObject;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;

public class NoteStorage {
    private static final String FILE_NAME = "notes_data.json";

    // Resolves dynamically to the standard Download/diet-tracker-data directory on any Android device
    private static final File TARGET_DIRECTORY = new File(
            Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS),
            "diet-tracker-data"
    );

    /**
     * Resolves the target directory and ensures the folder structure is created.
     */
    private static File getTargetFile() {
        if (!TARGET_DIRECTORY.exists()) {
            TARGET_DIRECTORY.mkdirs(); // Automatically creates "diet-tracker-data" if missing
        }
        return new File(TARGET_DIRECTORY, FILE_NAME);
    }

    /**
     * Saves the list of notes to the specified external storage path.
     */
    public static void saveNotes(Context context, List<Note> notes) {
        try {
            JSONArray jsonArray = new JSONArray();
            for (Note note : notes) {
                JSONObject noteJson = note.toJsonObject();
                if (noteJson != null) {
                    jsonArray.put(noteJson);
                }
            }
            String jsonContent = jsonArray.toString(4);

            File file = getTargetFile();
            try (FileOutputStream fos = new FileOutputStream(file);
                 OutputStreamWriter osw = new OutputStreamWriter(fos, StandardCharsets.UTF_8)) {
                osw.write(jsonContent);
            }
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error saving notes", e);
        }
    }

    /**
     * Loads the list of notes from the specified external storage path.
     */
    public static List<Note> loadNotes(Context context) {
        List<Note> notes = new ArrayList<>();
        File file = getTargetFile();

        if (!file.exists()) {
            return notes;
        }

        try (FileInputStream fis = new FileInputStream(file);
             InputStreamReader isr = new InputStreamReader(fis, StandardCharsets.UTF_8);
             BufferedReader reader = new BufferedReader(isr)) {

            StringBuilder sb = new StringBuilder();
            char[] buffer = new char[1024];
            int numRead;
            while ((numRead = reader.read(buffer)) != -1) {
                sb.append(buffer, 0, numRead);
            }

            String jsonContent = sb.toString();

            // Strip hidden UTF-8 BOM if present (often added by Windows/external text editors)
            if (jsonContent.startsWith("\uFEFF")) {
                jsonContent = jsonContent.substring(1);
            }
            jsonContent = jsonContent.trim();

            if (!jsonContent.isEmpty()) {
                JSONArray jsonArray = new JSONArray(jsonContent);
                for (int i = 0; i < jsonArray.length(); i++) {
                    JSONObject noteJson = jsonArray.getJSONObject(i);
                    Note note = Note.fromJsonObject(noteJson);
                    if (note != null) {
                        notes.add(note);
                    }
                }
            }
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error reading or parsing notes JSON file", e);
        }

        return notes;
    }

    /**
     * Imports external JSON files via Uri, validates structure, and overwrites local notes storage.
     */
    public static boolean importNotes(Context context, Uri sourceUri) {
        try (InputStream in = context.getContentResolver().openInputStream(sourceUri)) {
            if (in == null) return false;

            BufferedReader reader = new BufferedReader(new InputStreamReader(in, StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            char[] buffer = new char[1024];
            int numRead;
            while ((numRead = reader.read(buffer)) != -1) {
                sb.append(buffer, 0, numRead);
            }

            String jsonContent = sb.toString();

            // Strip hidden UTF-8 BOM if present
            if (jsonContent.startsWith("\uFEFF")) {
                jsonContent = jsonContent.substring(1);
            }
            jsonContent = jsonContent.trim();

            JSONArray jsonArray = new JSONArray(jsonContent);
            List<Note> importedNotes = new ArrayList<>();
            for (int i = 0; i < jsonArray.length(); i++) {
                Note note = Note.fromJsonObject(jsonArray.getJSONObject(i));
                if (note != null) {
                    importedNotes.add(note);
                }
            }

            // Write validated imported notes directly to the target external storage
            saveNotes(context, importedNotes);
            return true;
        } catch (Exception e) {
            android.util.Log.e("NoteStorage", "Error importing notes", e);
            return false;
        }
    }
}

public class ColorHueBarView extends View {
    private Paint paint;
    private Paint thumbPaint;
    private float hue = 0f;
    private OnHueChangedListener listener;

    private static final int[] RAINBOW = {
            0xFFFF0000, 0xFFFF00FF, 0xFF0000FF, 0xFF00FFFF, 0xFF00FF00, 0xFFFFFF00, 0xFFFF0000
    };

    public interface OnHueChangedListener {
        void onHueChanged(float hue);
    }

    public ColorHueBarView(Context context, AttributeSet attrs) {
        super(context, attrs);
        paint = new Paint(Paint.ANTI_ALIAS_FLAG);
        thumbPaint = new Paint(Paint.ANTI_ALIAS_FLAG);
        thumbPaint.setStyle(Paint.Style.STROKE);
        thumbPaint.setStrokeWidth(5f);
        thumbPaint.setColor(Color.WHITE);
    }

    public void setListener(OnHueChangedListener listener) {
        this.listener = listener;
    }

    public void setHue(float hue) {
        this.hue = Math.max(0f, Math.min(360f, hue));
        invalidate();
    }

    @Override
    protected void onSizeChanged(int w, int h, int oldw, int oldh) {
        super.onSizeChanged(w, h, oldw, oldh);
        LinearGradient gradient = new LinearGradient(0, 0, 0, h, RAINBOW, null, Shader.TileMode.CLAMP);
        paint.setShader(gradient);
    }

    @Override
    protected void onDraw(Canvas canvas) {
        super.onDraw(canvas);
        int w = getWidth();
        int h = getHeight();
        canvas.drawRect(0, 0, w, h, paint);

        float y = (hue / 360f) * h;
        canvas.drawRect(0, y - 4f, w, y + 4f, thumbPaint);
    }

    @Override
    public boolean onTouchEvent(MotionEvent event) {
        if (event.getAction() == MotionEvent.ACTION_DOWN || event.getAction() == MotionEvent.ACTION_MOVE) {
            float y = Math.max(0f, Math.min(getHeight(), event.getY()));
            hue = (y / getHeight()) * 360f;
            invalidate();
            if (listener != null) listener.onHueChanged(hue);
            return true;
        }
        return super.onTouchEvent(event);
    }
}

package com.lukaszjag.diet_tracker_android.tools.widget;

import android.content.Context;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.ComposeShader;
import android.graphics.LinearGradient;
import android.graphics.Paint;
import android.graphics.PorterDuff;
import android.graphics.Shader;
import android.util.AttributeSet;
import android.view.MotionEvent;
import android.view.View;

public class ColorSatValView extends View {
    private Paint paint;
    private Paint thumbPaint;
    private Shader valShader;
    private float hue = 0f;
    private float sat = 0f;
    private float val = 1f;
    private OnColorChangedListener listener;

    public interface OnColorChangedListener {
        void onSatValChanged(float sat, float val);
    }

    public ColorSatValView(Context context, AttributeSet attrs) {
        super(context, attrs);
        paint = new Paint(Paint.ANTI_ALIAS_FLAG);
        thumbPaint = new Paint(Paint.ANTI_ALIAS_FLAG);
        thumbPaint.setStyle(Paint.Style.STROKE);
        thumbPaint.setStrokeWidth(5f);
    }

    public void setListener(OnColorChangedListener listener) {
        this.listener = listener;
    }

    public void setHue(float hue) {
        this.hue = hue;
        updateShader();
        invalidate();
    }

    public void setSatVal(float sat, float val) {
        this.sat = Math.max(0f, Math.min(1f, sat));
        this.val = Math.max(0f, Math.min(1f, val));
        invalidate();
    }

    @Override
    protected void onSizeChanged(int w, int h, int oldw, int oldh) {
        super.onSizeChanged(w, h, oldw, oldh);
        valShader = new LinearGradient(0, 0, 0, h, Color.TRANSPARENT, Color.BLACK, Shader.TileMode.CLAMP);
        updateShader();
    }

    private void updateShader() {
        int w = getWidth();
        if (w <= 0 || valShader == null) return;
        int pureHue = Color.HSVToColor(new float[]{hue, 1f, 1f});
        Shader satShader = new LinearGradient(0, 0, w, 0, Color.WHITE, pureHue, Shader.TileMode.CLAMP);
        ComposeShader composeShader = new ComposeShader(satShader, valShader, PorterDuff.Mode.SRC_OVER);
        paint.setShader(composeShader);
    }

    @Override
    protected void onDraw(Canvas canvas) {
        super.onDraw(canvas);
        int w = getWidth();
        int h = getHeight();
        canvas.drawRect(0, 0, w, h, paint);

        float cx = sat * w;
        float cy = (1f - val) * h;
        thumbPaint.setColor(val > 0.5f ? Color.BLACK : Color.WHITE);
        canvas.drawCircle(cx, cy, 14f, thumbPaint);
    }

    @Override
    public boolean onTouchEvent(MotionEvent event) {
        if (event.getAction() == MotionEvent.ACTION_DOWN || event.getAction() == MotionEvent.ACTION_MOVE) {
            float x = Math.max(0f, Math.min(getWidth(), event.getX()));
            float y = Math.max(0f, Math.min(getHeight(), event.getY()));
            sat = x / getWidth();
            val = 1f - (y / getHeight());
            invalidate();
            if (listener != null) listener.onSatValChanged(sat, val);
            return true;
        }
        return super.onTouchEvent(event);
    }
}

import android.graphics.Color;
import android.net.Uri;
import android.os.Build;
import android.util.Log;
import android.widget.RemoteViews;

import com.lukaszjag.diet_tracker_android.R;
import com.lukaszjag.diet_tracker_android.gui.diet.day_data_view.DayData;

import java.util.Locale;

public class DietWidgetProvider extends AppWidgetProvider {

    private static final String TAG = "WIDGET_DEBUG";
    private static final String PREFS_NAME = "DietWidgetPrefs";
    private static final String PREF_PREFIX_KEY = "selected_date_";

    public static final String ACTION_REFRESH = "com.lukaszjag.diet_tracker_android.ACTION_WIDGET_REFRESH";
    public static final String ACTION_PREV_DAY = "com.lukaszjag.diet_tracker_android.ACTION_WIDGET_PREV_DAY";
    public static final String ACTION_NEXT_DAY = "com.lukaszjag.diet_tracker_android.ACTION_WIDGET_NEXT_DAY";

    // Pobranie aktualnie wybranej daty dla danego widżetu (domyślnie: dzisiaj)
    private static String getSelectedDate(Context context, int appWidgetId) {
        SharedPreferences prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE);
        String today = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).format(new Date());
        return prefs.getString(PREF_PREFIX_KEY + appWidgetId, today);
    }

    // Zapisanie wybranej daty
    private static void saveSelectedDate(Context context, int appWidgetId, String date) {
        SharedPreferences prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE);
        prefs.edit().putString(PREF_PREFIX_KEY + appWidgetId, date).apply();
    }

    @Override
    public void onUpdate(Context context, AppWidgetManager appWidgetManager, int[] appWidgetIds) {
        for (int widgetId : appWidgetIds) {
            fetchDataAndUpdateWidget(context, appWidgetManager, widgetId, null);
        }
    }

    @Override
    public void onReceive(Context context, Intent intent) {
        super.onReceive(context, intent);
        String action = intent.getAction();
        Log.d(TAG, "onReceive action: " + action);

        if (action == null) return;

        AppWidgetManager appWidgetManager = AppWidgetManager.getInstance(context);
        int appWidgetId = intent.getIntExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, AppWidgetManager.INVALID_APPWIDGET_ID);

        // Uruchomienie goAsync() zapobiega uśpieniu procesu podczas zapytania do Azure
        final PendingResult pendingResult = goAsync();

        if (ACTION_PREV_DAY.equals(action) && appWidgetId != AppWidgetManager.INVALID_APPWIDGET_ID) {
            String currentDate = getSelectedDate(context, appWidgetId);
            String prevDate = MyDate.getPreviousDayDateSQLFriendlyFormat(currentDate);
            saveSelectedDate(context, appWidgetId, prevDate);
            fetchDataAndUpdateWidget(context, appWidgetManager, appWidgetId, pendingResult);

        } else if (ACTION_NEXT_DAY.equals(action) && appWidgetId != AppWidgetManager.INVALID_APPWIDGET_ID) {
            String currentDate = getSelectedDate(context, appWidgetId);
            String nextDate = MyDate.getNextDayDateSQLFriendlyFormat(currentDate);
            saveSelectedDate(context, appWidgetId, nextDate);
            fetchDataAndUpdateWidget(context, appWidgetManager, appWidgetId, pendingResult);

        } else if (ACTION_REFRESH.equals(action) || AppWidgetManager.ACTION_APPWIDGET_UPDATE.equals(action)) {
            ComponentName thisWidget = new ComponentName(context, DietWidgetProvider.class);
            int[] allWidgetIds = appWidgetManager.getAppWidgetIds(thisWidget);

            if (allWidgetIds.length == 0) {
                if (pendingResult != null) pendingResult.finish();
                return;
            }

            for (int i = 0; i < allWidgetIds.length; i++) {
                PendingResult pr = (i == allWidgetIds.length - 1) ? pendingResult : null;
                fetchDataAndUpdateWidget(context, appWidgetManager, allWidgetIds[i], pr);
            }
        } else {
            if (pendingResult != null) pendingResult.finish();
        }
    }

    @Override
    public void onDeleted(Context context, int[] appWidgetIds) {
        super.onDeleted(context, appWidgetIds);
        // Czyszczenie preferencji po usunięciu widżetu
        SharedPreferences.Editor editor = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE).edit();
        for (int widgetId : appWidgetIds) {
            editor.remove(PREF_PREFIX_KEY + widgetId);
        }
        editor.apply();
    }

    private static void fetchDataAndUpdateWidget(Context context, AppWidgetManager appWidgetManager, int appWidgetId, PendingResult pendingResult) {
        RemoteViews views = new RemoteViews(context.getPackageName(), R.layout.widget_calorie_tracker);

        int flags = PendingIntent.FLAG_UPDATE_CURRENT;
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            flags |= PendingIntent.FLAG_IMMUTABLE;
        }

        Intent settingsIntent = new Intent(context, WidgetColorSettingsActivity.class);
        PendingIntent settingsPendingIntent = PendingIntent.getActivity(context, 999, settingsIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetSettingsButton, settingsPendingIntent);

        // 1. Kliknięcie widżetu -> otwarcie DayData
        Intent openAppIntent = new Intent(context, DayData.class);
        PendingIntent openAppPendingIntent = PendingIntent.getActivity(context, 0, openAppIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetRoot, openAppPendingIntent);

        // 2. Kliknięcie: Poprzedni dzień (<)
        Intent prevIntent = new Intent(context, DietWidgetProvider.class);
        prevIntent.setAction(ACTION_PREV_DAY);
        prevIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        prevIntent.setData(Uri.parse("dietwidget://prev/" + appWidgetId));
        PendingIntent prevPendingIntent = PendingIntent.getBroadcast(context, appWidgetId * 10 + 1, prevIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetPrevDayButton, prevPendingIntent);

        // 3. Kliknięcie: Następny dzień (>)
        Intent nextIntent = new Intent(context, DietWidgetProvider.class);
        nextIntent.setAction(ACTION_NEXT_DAY);
        nextIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        nextIntent.setData(Uri.parse("dietwidget://next/" + appWidgetId));
        PendingIntent nextPendingIntent = PendingIntent.getBroadcast(context, appWidgetId * 10 + 2, nextIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetNextDayButton, nextPendingIntent);

        // 4. Kliknięcie: Odśwież (⟳)
        Intent refreshIntent = new Intent(context, DietWidgetProvider.class);
        refreshIntent.setAction(ACTION_REFRESH);
        refreshIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        refreshIntent.setData(Uri.parse("dietwidget://refresh/" + appWidgetId));
        PendingIntent refreshPendingIntent = PendingIntent.getBroadcast(context, appWidgetId * 10 + 3, refreshIntent, flags);
        views.setOnClickPendingIntent(R.id.widgetRefreshButton, refreshPendingIntent);

        // 5. Podpięcie ListView
        Intent serviceIntent = new Intent(context, DietWidgetService.class);
        serviceIntent.putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, appWidgetId);
        serviceIntent.setData(Uri.parse(serviceIntent.toUri(Intent.URI_INTENT_SCHEME)));
        views.setRemoteAdapter(R.id.widgetListView, serviceIntent);
        views.setEmptyView(R.id.widgetListView, R.id.widgetEmptyView);

        // 6. Ustalenie daty do wyświetlenia
        String selectedDate = getSelectedDate(context, appWidgetId);
        String todayDate = new SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).format(new Date());

        String displayDateText;
        if (selectedDate.equals(todayDate)) {
            displayDateText = "Today (" + selectedDate.substring(5) + ")";
        } else {
            displayDateText = selectedDate;
        }

        views.setTextViewText(R.id.widgetDateTextView, displayDateText);
        views.setTextViewText(R.id.widgetKcalTextView, "Pobieranie...");
        appWidgetManager.partiallyUpdateAppWidget(appWidgetId, views);

        // 7. Zapytanie SQL do Azure
        String query = QueryMaker.getAllProductsFromCalendarTableLikeDate(selectedDate);
        Log.d(TAG, "Fetching data for date: " + selectedDate + " | Query: " + query);

        SharedPreferences prefs = context.getSharedPreferences("DietWidgetPrefs", Context.MODE_PRIVATE);
        int bgColor = prefs.getInt("widget_bg_color", Color.WHITE);
        views.setInt(R.id.widgetRoot, "setBackgroundColor", bgColor);

// Jeśli tło jest ciemne, zmień kolor daty na jasny
        double darkness = 1 - (0.299 * Color.red(bgColor) + 0.587 * Color.green(bgColor) + 0.114 * Color.blue(bgColor)) / 255;
        int headerTextColor = (darkness > 0.5) ? Color.WHITE : Color.parseColor("#333333");
        views.setTextColor(R.id.widgetDateTextView, headerTextColor);

        GetFromSQLDatabase.runAzureQueryAIChat(query, new AzureDataCallback() {
            @Override
            public void onSuccess(ArrayList<RowInTable> resultTable) {
                float totalKcal = 0;
                float totalProtein = 0;
                float totalFat = 0;
                float totalCarbs = 0;

                ArrayList<RowInTable> rows = (resultTable != null) ? resultTable : new ArrayList<>();

                for (RowInTable row : rows) {
                    totalKcal += safeParse(row.getValue("kcal_consume"));
                    totalProtein += safeParse(row.getValue("protein_consume"));
                    totalFat += safeParse(row.getValue("fat_consume"));
                    totalCarbs += safeParse(row.getValue("carbs_consume"));
                }

                views.setTextViewText(R.id.widgetKcalTextView, String.format(Locale.getDefault(), "%.0f kcal", totalKcal));
                views.setTextViewText(R.id.widgetProteinTextView, String.format(Locale.getDefault(), "P: %.1fg", totalProtein));
                views.setTextViewText(R.id.widgetFatTextView, String.format(Locale.getDefault(), "F: %.1fg", totalFat));
                views.setTextViewText(R.id.widgetCarbsTextView, String.format(Locale.getDefault(), "C: %.1fg", totalCarbs));

                DietWidgetService.productList = rows;
                appWidgetManager.notifyAppWidgetViewDataChanged(appWidgetId, R.id.widgetListView);
                appWidgetManager.updateAppWidget(appWidgetId, views);

                if (pendingResult != null) pendingResult.finish();
            }

            @Override
            public void onFailure(String errorMessage) {
                Log.e(TAG, "Fetch failed: " + errorMessage);
                views.setTextViewText(R.id.widgetKcalTextView, "Błąd sieci");
                appWidgetManager.updateAppWidget(appWidgetId, views);

                if (pendingResult != null) pendingResult.finish();
            }
        });
    }

    private static float safeParse(String val) {
        if (val == null || val.trim().isEmpty() || val.equalsIgnoreCase("null")) return 0f;
        try {
            return Float.parseFloat(val);
        } catch (NumberFormatException e) {
            return 0f;
        }
    }

    public static void forceWidgetUpdate(Context context) {
        Intent intent = new Intent(context, DietWidgetProvider.class);
        intent.setAction(ACTION_REFRESH);
        context.sendBroadcast(intent);
    }
}

package com.lukaszjag.diet_tracker_android.tools.widget;

import android.content.Context;
import android.content.Intent;
import android.widget.RemoteViews;
import android.widget.RemoteViewsService;

import com.lukaszjag.diet_tracker_android.R;
import com.lukaszjag.diet_tracker_android.tools.sql_tools.RowInTable;

import java.util.ArrayList;
import java.util.List;
import java.util.Locale;

public class DietWidgetService extends RemoteViewsService {

    // Lista produktów współdzielona z widżetem
    public static List<RowInTable> productList = new ArrayList<>();

    @Override
    public RemoteViewsFactory onGetViewFactory(Intent intent) {
        return new DietRemoteViewsFactory(this.getApplicationContext());
    }

    static class DietRemoteViewsFactory implements RemoteViewsFactory {
        private final Context context;

        public DietRemoteViewsFactory(Context context) {
            this.context = context;
        }

        @Override
        public void onCreate() {}

        @Override
        public void onDataSetChanged() {
            // Wywoływane po notifyAppWidgetViewDataChanged
        }

        @Override
        public void onDestroy() {}

        @Override
        public int getCount() {
            return productList != null ? productList.size() : 0;
        }

        @Override
        public RemoteViews getViewAt(int position) {
            if (productList == null || position < 0 || position >= productList.size()) {
                return null;
            }

            RowInTable row = productList.get(position);
            RemoteViews views = new RemoteViews(context.getPackageName(), R.layout.widget_product_item);

            String name = row.getValue("product_name");
            String meal = row.getValue("meal_name");
            String amountStr = row.getValue("amount_of_product");
            String kcalStr = row.getValue("kcal_consume");

            float amount = 0f;
            float kcal = 0f;
            try {
                if (amountStr != null) amount = Float.parseFloat(amountStr);
                if (kcalStr != null) kcal = Float.parseFloat(kcalStr);
            } catch (NumberFormatException ignored) {}

            views.setTextViewText(R.id.itemProductName, name != null ? name : "Unknown");

            String details = (meal != null ? meal : "Meal") + " • " + String.format(Locale.getDefault(), "%.0fg", amount);
            views.setTextViewText(R.id.itemProductDetails, details);

            views.setTextViewText(R.id.itemProductKcal, String.format(Locale.getDefault(), "%.0f kcal", kcal));

            return views;
        }

        @Override
        public RemoteViews getLoadingView() {
            return null;
        }

        @Override
        public int getViewTypeCount() {
            return 1;
        }

        @Override
        public long getItemId(int position) {
            return position;
        }

        @Override
        public boolean hasStableIds() {
            return true;
        }
    }
}

package com.lukaszjag.diet_tracker_android.tools.widget;

import android.content.Context;
import android.content.SharedPreferences;
import android.graphics.Color;
import android.os.Bundle;
import android.text.Editable;
import android.text.TextWatcher;
import android.widget.Button;
import android.widget.EditText;
import android.widget.ImageView;
import android.widget.Toast;

import androidx.annotation.Nullable;
import androidx.appcompat.app.AppCompatActivity;

import com.lukaszjag.diet_tracker_android.R;

import java.util.Locale;

public class WidgetColorSettingsActivity extends AppCompatActivity {

    private ColorSatValView satValPicker;
    private ColorHueBarView huePicker;
    private ImageView colorPreview;
    private EditText etR, etG, etB, etH, etS, etV, etHex;
    private Button saveColorButton;

    private float currentHue = 0f;
    private float currentSat = 0f;
    private float currentVal = 1f;
    private boolean isUpdatingFromCode = false;

    @Override
    protected void onCreate(@Nullable Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_widget_color_settings);

        satValPicker = findViewById(R.id.satValPicker);
        huePicker = findViewById(R.id.huePicker);
        colorPreview = findViewById(R.id.colorPreview);

        etR = findViewById(R.id.etR);
        etG = findViewById(R.id.etG);
        etB = findViewById(R.id.etB);
        etH = findViewById(R.id.etH);
        etS = findViewById(R.id.etS);
        etV = findViewById(R.id.etV);
        etHex = findViewById(R.id.etHex);
        saveColorButton = findViewById(R.id.saveColorButton);

        // Odczyt wcześniej zapisanego koloru (domyślnie biały)
        SharedPreferences prefs = getSharedPreferences("DietWidgetPrefs", Context.MODE_PRIVATE);
        int savedColor = prefs.getInt("widget_bg_color", Color.WHITE);
        setColor(savedColor);

        // Listenery pickera
        satValPicker.setListener((sat, val) -> {
            currentSat = sat;
            currentVal = val;
            updateColorViews();
        });

        huePicker.setListener(hue -> {
            currentHue = hue;
            satValPicker.setHue(hue);
            updateColorViews();
        });

        // Wpisywanie ręczne HEX
        etHex.addTextChangedListener(new TextWatcher() {
            @Override public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            @Override public void onTextChanged(CharSequence s, int start, int before, int count) {
                if (isUpdatingFromCode) return;
                try {
                    String hex = s.toString().trim();
                    if (hex.length() == 6) {
                        int color = Color.parseColor("#" + hex);
                        setColor(color);
                    }
                } catch (Exception ignored) {}
            }
            @Override public void afterTextChanged(Editable s) {}
        });

        // Szablony
        findViewById(R.id.presetWhite).setOnClickListener(v -> setColor(0xFFFFFFFF));
        findViewById(R.id.presetDark).setOnClickListener(v -> setColor(0xFF252525));
        findViewById(R.id.presetAmoled).setOnClickListener(v -> setColor(0xFF000000));
        findViewById(R.id.presetBlue).setOnClickListener(v -> setColor(0xFF1E3A8A));

        // Zapis
        saveColorButton.setOnClickListener(v -> {
            int finalColor = Color.HSVToColor(new float[]{currentHue, currentSat, currentVal});
            prefs.edit().putInt("widget_bg_color", finalColor).apply();

            // Odśwież widżet na pulpicie
            DietWidgetProvider.forceWidgetUpdate(this);

            Toast.makeText(this, "Widget background updated!", Toast.LENGTH_SHORT).show();
            finish();
        });
    }

    private void setColor(int color) {
        float[] hsv = new float[3];
        Color.colorToHSV(color, hsv);
        currentHue = hsv[0];
        currentSat = hsv[1];
        currentVal = hsv[2];

        huePicker.setHue(currentHue);
        satValPicker.setHue(currentHue);
        satValPicker.setSatVal(currentSat, currentVal);

        updateColorViews();
    }

    private void updateColorViews() {
        int color = Color.HSVToColor(new float[]{currentHue, currentSat, currentVal});
        colorPreview.setBackgroundColor(color);

        isUpdatingFromCode = true;

        etR.setText(String.valueOf(Color.red(color)));
        etG.setText(String.valueOf(Color.green(color)));
        etB.setText(String.valueOf(Color.blue(color)));

        etH.setText(String.format(Locale.getDefault(), "%.0f", currentHue));
        etS.setText(String.format(Locale.getDefault(), "%.0f", currentSat * 100));
        etV.setText(String.format(Locale.getDefault(), "%.0f", currentVal * 100));

        String hex = String.format("%06X", (0xFFFFFF & color));
        etHex.setText(hex);

        isUpdatingFromCode = false;
    }
}
