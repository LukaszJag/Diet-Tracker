Here is the complete solution implementing all your requirements.

Summary of Changes & Architecture

1.  App Preferences & Theme Helper:
      - AppSettings.java: Stores preferences for button background color,
        calendar heatmap base color, and calendar UI scale factor (Small,
        Medium, Large, Extra Large).
      - ThemeHelper.java: Traverses the UI hierarchy and dynamically applies the
        chosen button background color and high-contrast text color across
        MainActivity, Notes, DayData, etc.
2.  Settings Activity:
      - AppSettingsActivity.java & activity_app_settings.xml: Allows customizing
        the background color of buttons in the app, the base color for calendar
        note heatmaps (with presets and custom HEX input), and the size of
        calendar elements. Accessible via the Settings button in MainActivity
        and the toolbar options menu.
3.  Calendar View & Dialog:
      - NotesCalendarDialog.java, dialog_notes_calendar.xml, &
        item_calendar_day.xml:
          - Opens via the 📅 Date Filter button in Notes.
          - Starts at the current month with < (previous) and > (next) month
            navigation.
          - Every Sunday number is colored Dark Red (#8B0000).
          - Each day's background intensity depends on the number of notes
            created on that day: white (#FFFFFF) if 0 notes, and progressively
            more intensive (using the chosen heatmap color) as note count
            increases.
          - Days can be tapped to select/unselect. Selected days display an
            outline and checkmark.
          - "Uncheck all days" button at the bottom clears all day selections.
          - "Apply Filter" button applies the selected dates.
4.  Combined Filtering in Notes:
      - If no day is selected (or when cleared), all notes are displayed.
      - If days are selected, only notes created on those dates are shown,
        smoothly combined with Subtitle, Category, Subcategory, Learning,
        General, Today, and To Buy filters.
      - Updated Notes.java, MyAdapter.java, and notes_view_layout.xml.

package com.lukaszjag.diet_tracker_android.tools.settings;

import android.content.Context;
import android.content.SharedPreferences;
import android.graphics.Color;

public class AppSettings {
    private static final String PREF_NAME = "DietTrackerAppSettings";
    private static final String KEY_BUTTON_BG_COLOR = "app_button_bg_color";
    private static final String KEY_CALENDAR_HEATMAP_COLOR = "calendar_heatmap_color";
    private static final String KEY_CALENDAR_SCALE = "calendar_ui_scale";

    // Defaults
    public static final int DEFAULT_BUTTON_COLOR = Color.parseColor("#4253A6");
    public static final int DEFAULT_HEATMAP_COLOR = Color.parseColor("#4CAF50"); // Green
    public static final float DEFAULT_CALENDAR_SCALE = 1.0f; // 100%

    public static int getButtonColor(Context context) {
        SharedPreferences prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        return prefs.getInt(KEY_BUTTON_BG_COLOR, DEFAULT_BUTTON_COLOR);
    }

    public static void setButtonColor(Context context, int color) {
        SharedPreferences prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        prefs.edit().putInt(KEY_BUTTON_BG_COLOR, color).apply();
    }

    public static int getCalendarHeatmapColor(Context context) {
        SharedPreferences prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        return prefs.getInt(KEY_CALENDAR_HEATMAP_COLOR, DEFAULT_HEATMAP_COLOR);
    }

    public static void setCalendarHeatmapColor(Context context, int color) {
        SharedPreferences prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        prefs.edit().putInt(KEY_CALENDAR_HEATMAP_COLOR, color).apply();
    }

    public static float getCalendarScale(Context context) {
        SharedPreferences prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        return prefs.getFloat(KEY_CALENDAR_SCALE, DEFAULT_CALENDAR_SCALE);
    }

    public static void setCalendarScale(Context context, float scale) {
        SharedPreferences prefs = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        prefs.edit().putFloat(KEY_CALENDAR_SCALE, scale).apply();
    }
}

package com.lukaszjag.diet_tracker_android.tools.theme;

import android.app.Activity;
import android.graphics.Color;
import android.graphics.PorterDuff;
import android.graphics.drawable.Drawable;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;
import android.widget.TextView;

import com.lukaszjag.diet_tracker_android.tools.settings.AppSettings;

public class ThemeHelper {

    /**
     * Traverses the current Activity's view hierarchy and updates all Button backgrounds.
     */
    public static void applyButtonTheme(Activity activity) {
        if (activity == null) return;
        int buttonColor = AppSettings.getButtonColor(activity);
        View root = activity.findViewById(android.R.id.content);
        if (root != null) {
            applyThemeToViewHierarchy(root, buttonColor);
        }
    }

    public static void applyThemeToViewHierarchy(View view, int color) {
        if (view == null) return;
        if (view instanceof Button) {
            applyColorToButton((Button) view, color);
        } else if (view instanceof ViewGroup) {
            ViewGroup vg = (ViewGroup) view;
            for (int i = 0; i < vg.getChildCount(); i++) {
                applyThemeToViewHierarchy(vg.getChildAt(i), color);
            }
        }
    }

    public static void applyColorToButton(Button button, int color) {
        if (button == null) return;
        Drawable bg = button.getBackground();
        if (bg != null) {
            bg.mutate().setColorFilter(color, PorterDuff.Mode.SRC_IN);
        } else {
            button.setBackgroundColor(color);
        }

        // Maintain high contrast text (white or black depending on color darkness)
        double darkness = 1 - (0.299 * Color.red(color) + 0.587 * Color.green(color) + 0.114 * Color.blue(color)) / 255;
        button.setTextColor(darkness > 0.4 ? Color.WHITE : Color.BLACK);
    }
}

package com.lukaszjag.diet_tracker_android.gui.settings;

import android.graphics.Color;
import android.os.Bundle;
import android.text.Editable;
import android.text.TextWatcher;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.RadioGroup;
import android.widget.Toast;

import androidx.annotation.Nullable;
import androidx.appcompat.app.AppCompatActivity;

import com.lukaszjag.diet_tracker_android.R;
import com.lukaszjag.diet_tracker_android.tools.settings.AppSettings;
import com.lukaszjag.diet_tracker_android.tools.theme.ThemeHelper;

public class AppSettingsActivity extends AppCompatActivity {

    private View previewButtonColor;
    private EditText etButtonColorHex;
    private View previewHeatmapColor;
    private EditText etHeatmapColorHex;
    private RadioGroup rgCalendarSize;
    private Button btnSaveSettings;

    private int currentButtonColor;
    private int currentHeatmapColor;
    private float currentScale;
    private boolean isUpdatingFromCode = false;

    @Override
    protected void onCreate(@Nullable Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_app_settings);

        previewButtonColor = findViewById(R.id.previewButtonColor);
        etButtonColorHex = findViewById(R.id.etButtonColorHex);
        previewHeatmapColor = findViewById(R.id.previewHeatmapColor);
        etHeatmapColorHex = findViewById(R.id.etHeatmapColorHex);
        rgCalendarSize = findViewById(R.id.rgCalendarSize);
        btnSaveSettings = findViewById(R.id.btnSaveSettings);

        // Load existing settings
        currentButtonColor = AppSettings.getButtonColor(this);
        currentHeatmapColor = AppSettings.getCalendarHeatmapColor(this);
        currentScale = AppSettings.getCalendarScale(this);

        updateButtonColorUI(currentButtonColor);
        updateHeatmapColorUI(currentHeatmapColor);
        setupScaleRadio();
        setupListeners();
    }

    @Override
    protected void onResume() {
        super.onResume();
        ThemeHelper.applyButtonTheme(this);
    }

    private void updateButtonColorUI(int color) {
        currentButtonColor = color;
        previewButtonColor.setBackgroundColor(color);
        isUpdatingFromCode = true;
        etButtonColorHex.setText(String.format("%06X", (0xFFFFFF & color)));
        isUpdatingFromCode = false;
    }

    private void updateHeatmapColorUI(int color) {
        currentHeatmapColor = color;
        previewHeatmapColor.setBackgroundColor(color);
        isUpdatingFromCode = true;
        etHeatmapColorHex.setText(String.format("%06X", (0xFFFFFF & color)));
        isUpdatingFromCode = false;
    }

    private void setupScaleRadio() {
        if (currentScale <= 0.85f) {
            rgCalendarSize.check(R.id.rbSizeSmall);
        } else if (currentScale <= 1.05f) {
            rgCalendarSize.check(R.id.rbSizeMedium);
        } else if (currentScale <= 1.3f) {
            rgCalendarSize.check(R.id.rbSizeLarge);
        } else {
            rgCalendarSize.check(R.id.rbSizeExtraLarge);
        }
    }

    private void setupListeners() {
        // Button color presets
        findViewById(R.id.btnPresetNavy).setOnClickListener(v -> updateButtonColorUI(Color.parseColor("#4253A6")));
        findViewById(R.id.btnPresetGreen).setOnClickListener(v -> updateButtonColorUI(Color.parseColor("#2E7D32")));
        findViewById(R.id.btnPresetBlue).setOnClickListener(v -> updateButtonColorUI(Color.parseColor("#1565C0")));
        findViewById(R.id.btnPresetPurple).setOnClickListener(v -> updateButtonColorUI(Color.parseColor("#6A1B9A")));
        findViewById(R.id.btnPresetRed).setOnClickListener(v -> updateButtonColorUI(Color.parseColor("#B71C1C")));
        findViewById(R.id.btnPresetDark).setOnClickListener(v -> updateButtonColorUI(Color.parseColor("#37474F")));

        // Heatmap color presets
        findViewById(R.id.hmPresetGreen).setOnClickListener(v -> updateHeatmapColorUI(Color.parseColor("#4CAF50")));
        findViewById(R.id.hmPresetBlue).setOnClickListener(v -> updateHeatmapColorUI(Color.parseColor("#2196F3")));
        findViewById(R.id.hmPresetOrange).setOnClickListener(v -> updateHeatmapColorUI(Color.parseColor("#FF9800")));
        findViewById(R.id.hmPresetPurple).setOnClickListener(v -> updateHeatmapColorUI(Color.parseColor("#9C27B0")));
        findViewById(R.id.hmPresetRed).setOnClickListener(v -> updateHeatmapColorUI(Color.parseColor("#F44336")));

        // Text inputs
        etButtonColorHex.addTextChangedListener(new TextWatcher() {
            public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            public void onTextChanged(CharSequence s, int start, int before, int count) {
                if (isUpdatingFromCode) return;
                try {
                    String hex = s.toString().trim();
                    if (hex.length() == 6) {
                        currentButtonColor = Color.parseColor("#" + hex);
                        previewButtonColor.setBackgroundColor(currentButtonColor);
                    }
                } catch (Exception ignored) {}
            }
            public void afterTextChanged(Editable s) {}
        });

        etHeatmapColorHex.addTextChangedListener(new TextWatcher() {
            public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            public void onTextChanged(CharSequence s, int start, int before, int count) {
                if (isUpdatingFromCode) return;
                try {
                    String hex = s.toString().trim();
                    if (hex.length() == 6) {
                        currentHeatmapColor = Color.parseColor("#" + hex);
                        previewHeatmapColor.setBackgroundColor(currentHeatmapColor);
                    }
                } catch (Exception ignored) {}
            }
            public void afterTextChanged(Editable s) {}
        });

        // Save
        btnSaveSettings.setOnClickListener(v -> {
            int checkedId = rgCalendarSize.getCheckedRadioButtonId();
            if (checkedId == R.id.rbSizeSmall) currentScale = 0.8f;
            else if (checkedId == R.id.rbSizeMedium) currentScale = 1.0f;
            else if (checkedId == R.id.rbSizeLarge) currentScale = 1.25f;
            else if (checkedId == R.id.rbSizeExtraLarge) currentScale = 1.5f;

            AppSettings.setButtonColor(this, currentButtonColor);
            AppSettings.setCalendarHeatmapColor(this, currentHeatmapColor);
            AppSettings.setCalendarScale(this, currentScale);

            ThemeHelper.applyButtonTheme(this);
            Toast.makeText(this, "Settings saved successfully!", Toast.LENGTH_SHORT).show();
            finish();
        });
    }
}

    private void updateCalendar() {
        SimpleDateFormat sdf = new SimpleDateFormat("MMMM yyyy", Locale.getDefault());
        monthYearTextView.setText(sdf.format(currentDisplayCalendar.getTime()));

        // Calculate leading blanks (Monday-first: Mon=0 ... Sun=6)
        Calendar temp = (Calendar) currentDisplayCalendar.clone();
        temp.set(Calendar.DAY_OF_MONTH, 1);
        int firstDayOfWeek = temp.get(Calendar.DAY_OF_WEEK); // Sun=1, Mon=2
        int offset = (firstDayOfWeek == Calendar.SUNDAY) ? 6 : (firstDayOfWeek - Calendar.MONDAY);

        int daysInMonth = currentDisplayCalendar.getActualMaximum(Calendar.DAY_OF_MONTH);

        // Find max notes count in displayed month for proper gradient scaling
        int maxNotes = 1;
        int year = currentDisplayCalendar.get(Calendar.YEAR);
        int month = currentDisplayCalendar.get(Calendar.MONTH) + 1;
        for (int day = 1; day <= daysInMonth; day++) {
            String dateKey = String.format(Locale.getDefault(), "%04d-%02d-%02d", year, month, day);
            Integer count = notesCountPerDay.get(dateKey);
            if (count != null && count > maxNotes) {
                maxNotes = count;
            }
        }
        this.maxNotesCountInMonth = maxNotes;

        List<Integer> dayNumbers = new ArrayList<>();
        for (int i = 0; i < offset; i++) {
            dayNumbers.add(0); // empty cell placeholder
        }
        for (int day = 1; day <= daysInMonth; day++) {
            dayNumbers.add(day);
        }

        dayAdapter = new CalendarDayAdapter(dayNumbers);
        daysRecyclerView.setAdapter(dayAdapter);
        updateSelectedCountLabel();
    }

            selectedCountTextView.setText("No days selected (Showing all notes)");
        } else {
            selectedCountTextView.setText("Selected: " + selectedDates.size() + " day(s)");
        }
    }

    private class CalendarDayAdapter extends RecyclerView.Adapter<CalendarDayAdapter.DayViewHolder> {
        private final List<Integer> days;

        public CalendarDayAdapter(List<Integer> days) {

        }

        @NonNull
        @Override
        public DayViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
            View view = LayoutInflater.from(parent.getContext()).inflate(R.layout.item_calendar_day, parent, false);
            // Apply scale to cell height
            ViewGroup.LayoutParams lp = view.getLayoutParams();
            if (lp != null) {
                float density = getContext().getResources().getDisplayMetrics().density;
                lp.height = (int) (44 * density * uiScale);
                view.setLayoutParams(lp);
            }
            return new DayViewHolder(view);
        }

        @Override
        public void onBindViewHolder(@NonNull DayViewHolder holder, int position) {
            int dayNumber = days.get(position);
            if (dayNumber == 0) {
                holder.root.setVisibility(View.INVISIBLE);
                holder.root.setOnClickListener(null);
                return;
            }

            holder.root.setVisibility(View.VISIBLE);
            holder.tvDayNumber.setText(String.valueOf(dayNumber));
            holder.tvDayNumber.setTextSize(TypedValue.COMPLEX_UNIT_SP, 13 * uiScale);

            Calendar dayCal = (Calendar) currentDisplayCalendar.clone();
            dayCal.set(Calendar.DAY_OF_MONTH, dayNumber);
            boolean isSunday = (dayCal.get(Calendar.DAY_OF_WEEK) == Calendar.SUNDAY);

            // Sunday number MUST be dark red (#8B0000)
            if (isSunday) {
                holder.tvDayNumber.setTextColor(Color.parseColor("#8B0000"));
                holder.tvDayNumber.setTypeface(null, Typeface.BOLD);
            } else {
                holder.tvDayNumber.setTextColor(Color.parseColor("#222222"));
                holder.tvDayNumber.setTypeface(null, Typeface.NORMAL);
            }

            // Date string yyyy-MM-dd
            int year = currentDisplayCalendar.get(Calendar.YEAR);
            int month = currentDisplayCalendar.get(Calendar.MONTH) + 1;
            String dateKey = String.format(Locale.getDefault(), "%04d-%02d-%02d", year, month, dayNumber);

            // Note count heatmap calculation
            int noteCount = 0;
            if (notesCountPerDay.containsKey(dateKey)) {
                noteCount = notesCountPerDay.get(dateKey);
            }

            int backgroundColor;
            if (noteCount == 0) {
                // White if none
                backgroundColor = Color.WHITE;
            } else {
                // More intensive color as notes increase
                float intensity = 0.25f + 0.75f * ((float) noteCount / Math.max(1, maxNotesCountInMonth));
                backgroundColor = ColorUtils.blendARGB(Color.WHITE, heatmapBaseColor, Math.min(1.0f, intensity));
            }

            // Selection state
            boolean isSelected = selectedDates.contains(dateKey);
            GradientDrawable borderDrawable = new GradientDrawable();
            borderDrawable.setColor(backgroundColor);

            float density = getContext().getResources().getDisplayMetrics().density;
            if (isSelected) {
                borderDrawable.setStroke((int) (2.5f * density), Color.parseColor("#1565C0"));
                holder.ivCheck.setVisibility(View.VISIBLE);
            } else {
                borderDrawable.setStroke((int) (0.5f * density), Color.parseColor("#E0E0E0"));
                holder.ivCheck.setVisibility(View.GONE);
            }
            holder.root.setBackground(borderDrawable);

            // Toggle date selection on click
            holder.root.setOnClickListener(v -> {
                if (selectedDates.contains(dateKey)) {
                    selectedDates.remove(dateKey);
                } else {
                    selectedDates.add(dateKey);
                }
                notifyItemChanged(position);
                updateSelectedCountLabel();
            });
        }

        @Override
        public int getItemCount() {
            return days.size();
        }

public class MyAdapter extends RecyclerView.Adapter<MyAdapter.MyViewHolder> {

    public static final int SORT_NONE = 0;
    public static final int SORT_DATE_NEWEST = 1;
    public static final int SORT_DATE_OLDEST = 2;

    }

    public void setOnItemClickListener(OnItemClickListener listener) { this.listener = listener; }

    public void setSectionVisibilities(boolean hideS1, boolean hideS2, boolean hideS3, boolean hideS4) {
        this.hideSection1 = hideS1;
        this.hideSection2 = hideS2;
        this.hideSection3 = hideS3;
        this.hideSection4 = hideS4;

    }

        notifyDataSetChanged();
    }

    public void addItem(Note note) {
        noteList.add(note);
        originalList.add(note);
        applySort();
        notifyItemInserted(noteList.indexOf(note));
    }

    public void setItem(int position, Note note) {
        if (position >= 0 && position < noteList.size()) {
            Note oldNote = noteList.get(position);
            int origIdx = originalList.indexOf(oldNote);
            if (origIdx != -1) originalList.set(origIdx, note);
            noteList.set(position, note);
            applySort();
            notifyDataSetChanged();
        }
    }

    public void deleteItem(int position) {
        if (position >= 0 && position < noteList.size()) {
            Note note = noteList.get(position);
            originalList.remove(note);
            noteList.remove(position);
            notifyItemRemoved(position);
            notifyItemRangeChanged(position, noteList.size());
        }
    }

    private void applySort() {
        if (sortCriteria == SORT_NONE) return;
        Collections.sort(noteList, (n1, n2) -> {
            switch (sortCriteria) {
                case SORT_DATE_NEWEST:
                    return (n2.getDateCreated() != null ? n2.getDateCreated() : "")
                            .compareTo(n1.getDateCreated() != null ? n1.getDateCreated() : "");
                case SORT_DATE_OLDEST:
                    return (n1.getDateCreated() != null ? n1.getDateCreated() : "")
                            .compareTo(n2.getDateCreated() != null ? n2.getDateCreated() : "");
                case SORT_URGENCY_HIGH:
                    return Integer.compare(getUrgencyWeight(n2), getUrgencyWeight(n1));
                case SORT_URGENCY_LOW:
                    return Integer.compare(getUrgencyWeight(n1), getUrgencyWeight(n2));
                default:
                    return 0;
            }
        });
    }

    private int getUrgencyWeight(Note note) {
        if (note == null) return 0;
        int idx = note.getUrgentScaleEnglish().indexOf(note.getNoteUrgently());
        return (idx != -1) ? idx : 0;
    }

    public void filter(String subtitle, String selectedMainCategory, String selectedSubcategory,
                       boolean showLearningOnly, boolean showGeneralOnly, boolean showTodayOnly, boolean showToBuyOnly,

    }

    public void filter(String subtitle, String selectedMainCategory, String selectedSubcategory,
                       boolean showLearningOnly, boolean showGeneralOnly, boolean showTodayOnly, boolean showToBuyOnly,
                       String statusFilter, Set<String> selectedDates) {
        noteList.clear();
        String qSub = (subtitle != null) ? subtitle.toLowerCase().trim() : "";
        boolean isAllCategories = (selectedMainCategory == null || selectedMainCategory.equals("All Categories"));
        boolean isAllSubcategories = (selectedSubcategory == null || selectedSubcategory.equals("All Subcategories"));

        boolean hasDateFilter = (selectedDates != null && !selectedDates.isEmpty());

        for (Note note : originalList) {
            boolean matchSub = qSub.isEmpty() || (note.getNoteSubtitle() != null && note.getNoteSubtitle().toLowerCase().contains(qSub));

            boolean matchCat = true;
            if (!isAllCategories) {
                matchCat = note.getNoteCategory() != null && note.getNoteCategory().equalsIgnoreCase(selectedMainCategory.trim());
                if (matchCat && !isAllSubcategories) {
                    matchCat = note.getNoteSubcategory() != null && note.getNoteSubcategory().equalsIgnoreCase(selectedSubcategory.trim());
                }
            }

            boolean matchLearning = !showLearningOnly || note.isLearning();
            boolean matchGeneral = !showGeneralOnly || note.isGeneralToDo();
            boolean matchToday = !showTodayOnly || note.isTodayTask();
            boolean matchToBuy = !showToBuyOnly || note.isToBuyTask();

            // Status filter for history
            boolean matchStatus = true;
            if ("DONE".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDone();
            } else if ("DELETED".equalsIgnoreCase(statusFilter)) {
                matchStatus = note.isDeleted();
            }

            // Filter by creation date
            boolean matchDate = true;
            if (hasDateFilter) {
                String created = note.getDateCreated();
                if (created != null && created.length() >= 10) {
                    created = created.substring(0, 10);
                }
                matchDate = (created != null && selectedDates.contains(created));
            }

            if (matchSub && matchCat && matchLearning && matchGeneral && matchToday && matchToBuy && matchStatus && matchDate) {
                noteList.add(note);
            }
        }
        applySort();
        notifyDataSetChanged();
    }

    @NonNull

import com.lukaszjag.diet_tracker_android.gui.notes.NotesHistory;
import com.lukaszjag.diet_tracker_android.gui.settings.AppSettingsActivity;
import com.lukaszjag.diet_tracker_android.tools.theme.ThemeHelper;

public class MainActivity extends AppCompatActivity {

    private Button addMealButton;
    private Button getDayDataButton;
    private Button noteTasksButton;
    private Button notesHistoryButton;
    private Button appSettingsButton;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.main_window);

        setupUIComponents();
        addListeners();
    }

    @Override
    protected void onResume() {
        super.onResume();
        // Dynamically apply background color of all UI buttons
        ThemeHelper.applyButtonTheme(this);
    }

    private void setupUIComponents() {
        addMealButton = findViewById(R.id.addMealButton);
        getDayDataButton = findViewById(R.id.getDayDataButton);
        noteTasksButton = findViewById(R.id.noteTaskButtonButton);
        notesHistoryButton = findViewById(R.id.notesHistoryButton);
        appSettingsButton = findViewById(R.id.appSettingsButton);
    }

    private void addListeners() {
        noteTasksButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, Notes.class);
            startActivity(intent);
        });

        notesHistoryButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, NotesHistory.class);
            startActivity(intent);
        });

        addMealButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, GetProductData.class);
            startActivity(intent);
        });

        getDayDataButton.setOnClickListener(v -> {
            Intent intent = new Intent(MainActivity.this, DayData.class);
            startActivity(intent);
        });

        if (appSettingsButton != null) {
            appSettingsButton.setOnClickListener(v -> {
                Intent intent = new Intent(MainActivity.this, AppSettingsActivity.class);
                startActivity(intent);
            });
        }
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        getMenuInflater().inflate(R.menu.menu_main, menu);
        return true;
    }

    @Override
    public boolean onOptionsItemSelected(MenuItem item) {
        if (item.getItemId() == R.id.action_settings) {
            Intent intent = new Intent(MainActivity.this, AppSettingsActivity.class);
            startActivity(intent);
            return true;
        }
        return super.onOptionsItemSelected(item);
    }
}
